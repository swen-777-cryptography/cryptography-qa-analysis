331 #[getter] 

332 fn extensions(&self, py: pyo3::Python<'_>) -> pyo3::PyResult<pyo3::Py<pyo3::PyAny>> { 

333 let tbs_cert_list = &self.owned.borrow_dependent().tbs_cert_list; 

334  

335 x509::parse_and_cache_extensions( 

336 py, 

337 &self.cached_extensions, 

338 &tbs_cert_list.raw_crl_extensions, 

339 |ext| match ext.extn_id { 

340 oid::CRL_NUMBER_OID => { 

341 let bignum = ext.value::<asn1::BigUint<'_>>()?; 

342 let pynum = big_byte_slice_to_py_int(py, bignum.as_bytes())?; 

343 Ok(Some(types::CRL_NUMBER.get(py)?.call1((pynum,))?)) 

344 } 

345 oid::DELTA_CRL_INDICATOR_OID => { 

346 let bignum = ext.value::<asn1::BigUint<'_>>()?; 

347 let pynum = big_byte_slice_to_py_int(py, bignum.as_bytes())?; 

348 Ok(Some(types::DELTA_CRL_INDICATOR.get(py)?.call1((pynum,))?)) 

349 } 

350 oid::ISSUER_ALTERNATIVE_NAME_OID => { 

351 let gn_seq = ext.value::<IssuerAlternativeName<'_>>()?; 

352 let ians = x509::parse_general_names(py, &gn_seq)?; 

353 Ok(Some( 

354 types::ISSUER_ALTERNATIVE_NAME.get(py)?.call1((ians,))?, 

355 )) 

356 } 

357 oid::AUTHORITY_INFORMATION_ACCESS_OID => { 

358 let ads = certificate::parse_access_descriptions(py, ext)?; 

359 Ok(Some( 

360 types::AUTHORITY_INFORMATION_ACCESS.get(py)?.call1((ads,))?, 

361 )) 

362 } 

363 oid::AUTHORITY_KEY_IDENTIFIER_OID => { 

364 Ok(Some(certificate::parse_authority_key_identifier(py, ext)?)) 

365 } 

366 oid::ISSUING_DISTRIBUTION_POINT_OID => { 

367 let idp = ext.value::<crl::IssuingDistributionPoint<'_, Asn1Read>>()?; 

368 let (full_name, relative_name) = match idp.distribution_point { 

369 Some(data) => certificate::parse_distribution_point_name(py, data)?, 

370 None => (py.None().into_bound(py), py.None().into_bound(py)), 

371 }; 

372 let py_reasons = if let Some(reasons) = idp.only_some_reasons { 

373 certificate::parse_distribution_point_reasons(py, Some(&reasons))? 

374 } else { 

375 py.None().into_bound(py) 

376 }; 

377 Ok(Some(types::ISSUING_DISTRIBUTION_POINT.get(py)?.call1(( 

378 full_name, 

379 relative_name, 

380 idp.only_contains_user_certs, 

381 idp.only_contains_ca_certs, 

382 py_reasons, 

383 idp.indirect_crl, 

384 idp.only_contains_attribute_certs, 

385 ))?)) 

386 } 

387 oid::FRESHEST_CRL_OID => { 

388 let dp = certificate::parse_distribution_points(py, ext)?; 

389 Ok(Some(types::FRESHEST_CRL.get(py)?.call1((dp,))?)) 

390 } 

391 _ => Ok(None), 

392 }, 

393 ) 

394 }

652pub(crate) fn parse_crl_reason_flags<'p>( 

653 py: pyo3::Python<'p>, 

654 reason: &crl::CRLReason, 

655) -> CryptographyResult<pyo3::Bound<'p, pyo3::PyAny>> { 

656 let flag_name = match reason.value() { 

657 0 => "unspecified", 

658 1 => "key_compromise", 

659 2 => "ca_compromise", 

660 3 => "affiliation_changed", 

661 4 => "superseded", 

662 5 => "cessation_of_operation", 

663 6 => "certificate_hold", 

664 8 => "remove_from_crl", 

665 9 => "privilege_withdrawn", 

666 10 => "aa_compromise", 

667 value => { 

668 return Err(CryptographyError::from( 

669 pyo3::exceptions::PyValueError::new_err(format!( 

670 "Unsupported reason code: {value}" 

671 )), 

672 )) 

673 } 

674 }; 

675 Ok(types::REASON_FLAGS.get(py)?.getattr(flag_name)?) 

676}

678pub fn parse_crl_entry_ext<'p>( 

679 py: pyo3::Python<'p>, 

680 ext: &Extension<'p>, 

681) -> CryptographyResult<Option<pyo3::Bound<'p, pyo3::PyAny>>> { 

682 match ext.extn_id { 

683 oid::CRL_REASON_OID => { 

684 let flags = parse_crl_reason_flags(py, &ext.value::<crl::CRLReason>()?)?; 

685 Ok(Some(types::CRL_REASON.get(py)?.call1((flags,))?)) 

686 } 

687 oid::CERTIFICATE_ISSUER_OID => { 

688 let gn_seq = ext.value::<asn1::SequenceOf<'_, name::GeneralName<'_>>>()?; 

689 let gns = x509::parse_general_names(py, &gn_seq)?; 

690 Ok(Some(types::CERTIFICATE_ISSUER.get(py)?.call1((gns,))?)) 

691 } 

692 oid::INVALIDITY_DATE_OID => { 

693 let time = ext.value::<asn1::X509GeneralizedTime>()?; 

694 let py_dt = x509::datetime_to_py(py, time.as_datetime())?; 

695 Ok(Some(types::INVALIDITY_DATE.get(py)?.call1((py_dt,))?)) 

696 } 

697 _ => Ok(None), 

698 } 

699}

741#[pyo3::pyfunction] 

742pub(crate) fn create_x509_crl( 

743 py: pyo3::Python<'_>, 

744 builder: &pyo3::Bound<'_, pyo3::PyAny>, 

745 private_key: &pyo3::Bound<'_, pyo3::PyAny>, 

746 hash_algorithm: &pyo3::Bound<'_, pyo3::PyAny>, 

747 rsa_padding: &pyo3::Bound<'_, pyo3::PyAny>, 

748 ecdsa_deterministic: Option<bool>, 

749) -> CryptographyResult<CertificateRevocationList> { 

750 let sigalg = x509::sign::compute_signature_algorithm( 

751 py, 

752 private_key.to_owned(), 

753 hash_algorithm.to_owned(), 

754 rsa_padding.to_owned(), 

755 )?; 

756 let ka_vec = cryptography_keepalive::KeepAlive::new(); 

757 let ka_bytes = cryptography_keepalive::KeepAlive::new(); 

758  

759 let py_revoked_certs: Vec<pyo3::Bound<'_, RevokedCertificate>> = builder 

760 .getattr(pyo3::intern!(py, "_revoked_certificates"))? 

761 .extract()?; 

762 let revoked_certs: Vec<RawRevokedCertificate<'_>> = py_revoked_certs 

763 .iter() 

764 .map(|c| c.get().owned.borrow_dependent().clone()) 

765 .collect(); 

766  

767 let ka = cryptography_keepalive::KeepAlive::new(); 

768  

769 let py_issuer_name = builder.getattr(pyo3::intern!(py, "_issuer_name"))?; 

770 let py_this_update = builder 

771 .getattr(pyo3::intern!(py, "_last_update"))? 

772 .extract()?; 

773 let py_next_update = builder 

774 .getattr(pyo3::intern!(py, "_next_update"))? 

775 .extract()?; 

776 let tbs_cert_list = crl::TBSCertList { 

777 version: Some(1), 

778 signature: sigalg.clone(), 

779 issuer: x509::common::encode_name(py, &ka, &py_issuer_name)?, 

780 this_update: x509::certificate::time_from_py(py, &py_this_update)?, 

781 next_update: Some(x509::certificate::time_from_py(py, &py_next_update)?), 

782 revoked_certificates: if revoked_certs.is_empty() { 

783 None 

784 } else { 

785 Some(common::Asn1ReadableOrWritable::new_write( 

786 asn1::SequenceOfWriter::new(revoked_certs), 

787 )) 

788 }, 

789 raw_crl_extensions: x509::common::encode_extensions( 

790 py, 

791 &ka_vec, 

792 &ka_bytes, 

793 &builder.getattr(pyo3::intern!(py, "_extensions"))?, 

794 extensions::encode_extension, 

795 )?, 

796 }; 

797  

798 let tbs_bytes = asn1::write_single(&tbs_cert_list)?; 

799 let signature = x509::sign::sign_data( 

800 py, 

801 private_key.clone(), 

802 hash_algorithm.clone(), 

803 rsa_padding.clone(), 

804 ecdsa_deterministic, 

805 &tbs_bytes, 

806 )?; 

807 let data = asn1::write_single(&crl::CertificateRevocationList { 

808 tbs_cert_list, 

809 signature_algorithm: sigalg, 

810 signature_value: asn1::BitString::new(&signature, 0).unwrap(), 

811 })?; 

812 load_der_x509_crl(py, pyo3::types::PyBytes::new(py, &data).unbind(), None) 

813}
