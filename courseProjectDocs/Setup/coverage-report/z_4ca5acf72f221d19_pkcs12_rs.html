9pub fn kdf( 

10 pass: &str, 

11 salt: &[u8], 

12 id: u8, 

13 rounds: u64, 

14 key_len: usize, 

15 hash_alg: openssl::hash::MessageDigest, 

16) -> Result<Vec<u8>, openssl::error::ErrorStack> { 

17 // Encode the password as big-endian UTF-16 with NUL trailer 

18 let pass = pass 

19 .encode_utf16() 

20 .chain([0]) 

21 .flat_map(|v| v.to_be_bytes()) 

22 .collect::<Vec<u8>>(); 

23  

24 // Comments are borrowed from BoringSSL. 

25 // In the spec, |block_size| is called "v", but measured in bits. 

26 let block_size = hash_alg.block_size(); 

27  

28 // 1. Construct a string, D (the "diversifier"), by concatenating v/8 copies 

29 // of ID. 

30 let d = vec![id; block_size]; 

31  

32 // 2. Concatenate copies of the salt together to create a string S of length 

33 // v(ceiling(s/v)) bits (the final copy of the salt may be truncated to 

34 // create S). Note that if the salt is the empty string, then so is S. 

35 // 

36 // 3. Concatenate copies of the password together to create a string P of 

37 // length v(ceiling(p/v)) bits (the final copy of the password may be 

38 // truncated to create P). Note that if the password is the empty string, 

39 // then so is P. 

40 // 

41 // 4. Set I=S||P to be the concatenation of S and P. 

42 let s_len = block_size * salt.len().div_ceil(block_size); 

43 let p_len = block_size * pass.len().div_ceil(block_size); 

44  

45 let mut init_key = vec![0; s_len + p_len]; 

46 for i in 0..s_len { 

47 init_key[i] = salt[i % salt.len()]; 

48 } 

49 for i in 0..p_len { 

50 init_key[i + s_len] = pass[i % pass.len()]; 

51 } 

52  

53 let mut result = vec![0; key_len]; 

54 let mut pos = 0; 

55 loop { 

56 // A. Set A_i=H^r(D||I). (i.e., the r-th hash of D||I, 

57 // H(H(H(... H(D||I)))) 

58  

59 let mut h = openssl::hash::Hasher::new(hash_alg)?; 

60 h.update(&d)?; 

61 h.update(&init_key)?; 

62 let mut a = h.finish()?; 

63  

64 for _ in 1..rounds { 

65 let mut h = openssl::hash::Hasher::new(hash_alg)?; 

66 h.update(&a)?; 

67 a = h.finish()?; 

68 } 

69  

70 let to_add = a.len().min(result.len() - pos); 

71 result[pos..pos + to_add].copy_from_slice(&a[..to_add]); 

72 pos += to_add; 

73 if pos == result.len() { 

74 break; 

75 } 

76  

77 // B. Concatenate copies of A_i to create a string B of length v bits (the 

78 // final copy of A_i may be truncated to create B). 

79 let mut b = vec![0; block_size]; 

80 for i in 0..block_size { 

81 b[i] = a[i % a.len()]; 

82 } 

83  

84 // C. Treating I as a concatenation I_0, I_1, ..., I_(k-1) of v-bit blocks, 

85 // where k=ceiling(s/v)+ceiling(p/v), modify I by setting I_j=(I_j+B+1) mod 

86 // 2^v for each j. 

87 assert!(init_key.len() % block_size == 0); 

88 let mut j = 0; 

89 while j < init_key.len() { 

90 let mut carry = 1u16; 

91 let mut k = block_size - 1; 

92 loop { 

93 carry += init_key[k + j] as u16 + b[k] as u16; 

94 init_key[j + k] = carry as u8; 

95 carry >>= 8; 

96 if k == 0 { 

97 break; 

98 } 

99 k -= 1; 

100 } 

101 j += block_size; 

102 } 

103 } 

104  

105 Ok(result) 

106}
