32fn parse_name_value_tags(rdns: &Name<'_>) -> Vec<u8> { 

33 let mut tags = vec![]; 

34 for rdn in rdns.unwrap_read().clone() { 

35 let mut attributes = rdn.collect::<Vec<_>>(); 

36 assert_eq!(attributes.len(), 1); 

37  

38 tags.push(attributes.pop().unwrap().value.tag().as_u8().unwrap()); 

39 } 

40 tags 

41}

43fn time_tag(t: &Time) -> u8 { 

44 match t { 

45 Time::UtcTime(_) => asn1::UtcTime::TAG.as_u8().unwrap(), 

46 Time::GeneralizedTime(_) => asn1::GeneralizedTime::TAG.as_u8().unwrap(), 

47 } 

48}

51fn test_parse_certificate(data: &[u8]) -> CryptographyResult<TestCertificate> {

63#[pyo3::pyfunction] 

64#[pyo3(signature = (encoding, sig, msg, certs, options))] 

65fn pkcs7_verify( 

66 py: pyo3::Python<'_>, 

67 encoding: crate::serialization::Encoding, 

68 sig: &[u8], 

69 msg: Option<CffiBuf<'_>>, 

70 certs: Vec<pyo3::Py<PyCertificate>>, 

71 options: pyo3::Bound<'_, pyo3::types::PyList>, 

72) -> CryptographyResult<()> { 

73 let p7 = match encoding { 

74 crate::serialization::Encoding::DER => openssl::pkcs7::Pkcs7::from_der(sig)?, 

75 crate::serialization::Encoding::PEM => openssl::pkcs7::Pkcs7::from_pem(sig)?, 

76 _ => openssl::pkcs7::Pkcs7::from_smime(sig)?.0, 

77 }; 

78  

79 let mut flags = openssl::pkcs7::Pkcs7Flags::empty(); 

80 if options.contains(types::PKCS7_TEXT.get(py)?)? { 

81 flags |= openssl::pkcs7::Pkcs7Flags::TEXT; 

82 } 

83  

84 let store = { 

85 let mut b = openssl::x509::store::X509StoreBuilder::new()?; 

86 for cert in &certs { 

87 let der = asn1::write_single(cert.get().raw.borrow_dependent())?; 

88 b.add_cert(openssl::x509::X509::from_der(&der)?)?; 

89 } 

90 b.build() 

91 }; 

92 let certs = openssl::stack::Stack::new()?; 

93  

94 p7.verify( 

95 &certs, 

96 &store, 

97 msg.as_ref().map(|m| m.as_bytes()), 

98 None, 

99 flags, 

100 )?; 

101  

102 Ok(()) 

103}
