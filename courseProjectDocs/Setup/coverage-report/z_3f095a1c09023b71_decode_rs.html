18fn read_value<'a, T: asn1::SimpleAsn1Readable<'a>>( 

19 parser: &mut Parser<'a>, 

20 encoding: &Option<pyo3::Py<Encoding>>, 

21) -> ParseResult<T> { 

22 let value = match encoding { 

23 Some(e) => match e.get() { 

24 Encoding::Implicit(n) => parser.read_implicit_element::<T>(*n), 

25 Encoding::Explicit(n) => parser.read_explicit_element::<T>(*n), 

26 }, 

27 None => parser.read_element::<T>(), 

28 }?; 

29 Ok(value) 

30}

122fn decode_generalized_time<'a>( 

123 py: pyo3::Python<'a>, 

124 parser: &mut Parser<'a>, 

125 encoding: &Option<pyo3::Py<Encoding>>, 

126) -> ParseResult<pyo3::Bound<'a, GeneralizedTime>> { 

127 let value = read_value::<asn1::GeneralizedTime>(parser, encoding)?; 

128 let dt = value.as_datetime(); 

129  

130 let microseconds = match value.nanoseconds() { 

131 Some(x) if x % 1_000 == 0 => x / 1_000, 

132 Some(_) => { 

133 return Err(CryptographyError::Py( 

134 pyo3::exceptions::PyValueError::new_err( 

135 "decoded GeneralizedTime data has higher precision than supported", 

136 ), 

137 )) 

138 } 

139 None => 0, 

140 }; 

141  

142 let inner = crate::x509::datetime_to_py_utc_with_microseconds(py, dt, microseconds)? 

143 .cast_into::<pyo3::types::PyDateTime>()? 

144 .unbind(); 

145  

146 Ok(pyo3::Bound::new(py, GeneralizedTime { inner })?) 

147}

170 parser: &mut Parser<'a>, 

171 encoding: &Option<pyo3::Py<Encoding>>, 

172) -> ParseResult<pyo3::Bound<'a, Tlv>> {

184 }?; 

185 Ok(pyo3::Bound::new( 

186 py, 

187 Tlv { 

188 tag_bytes: pyo3::types::PyBytes::new(py, tlv.tag_bytes()).unbind(), 

189 data_index: tlv.full_data().len() - tlv.data().len(), 

190 full_data: pyo3::types::PyBytes::new(py, tlv.full_data()).unbind(), 

191 }, 

192 )?) 

193}

266// Decodes a value set field: decodes the underlying value, then maps 

267// it back to the enum member with that value. Fails if the decoded 

268// value does not correspond to any member. 

269fn decode_value_set<'a>( 

270 py: pyo3::Python<'a>, 

271 parser: &mut Parser<'a>, 

272 cls: &pyo3::Py<pyo3::types::PyType>, 

273 inner_type: &AnnotatedType, 

274 value_map: &pyo3::Py<pyo3::types::PyDict>, 

275 annotation: &Annotation, 

276) -> ParseResult<pyo3::Bound<'a, pyo3::PyAny>> { 

277 let inner_ann_type = value_set_inner_type(py, inner_type, annotation)?; 

278 let decoded = decode_annotated_type(py, parser, &inner_ann_type)?; 

279 if let Some(member) = value_map.bind(py).get_item(&decoded)? { 

280 return Ok(member); 

281 } 

282 Err(CryptographyError::Py( 

283 pyo3::exceptions::PyValueError::new_err(format!( 

284 "{} is not a valid value for {}", 

285 decoded.repr()?, 

286 cls.bind(py).name()?, 

287 )), 

288 )) 

289}

291// Utility function to handle explicit encoding when parsing 

292// CHOICE fields. 

293fn decode_choice_with_encoding<'a>( 

294 py: pyo3::Python<'a>, 

295 parser: &mut Parser<'a>, 

296 ann_type: &AnnotatedType, 

297 encoding: &Encoding, 

298) -> ParseResult<pyo3::Bound<'a, pyo3::PyAny>> { 

299 match encoding { 

300 Encoding::Implicit(_) => Err(CryptographyError::Py( 

301 // CHOICEs cannot be IMPLICIT. See X.680 section 31.2.9. 

302 pyo3::exceptions::PyValueError::new_err( 

303 "invalid type definition: CHOICE fields cannot be implicitly encoded", 

304 ), 

305 ))?, 

306 Encoding::Explicit(n) => { 

307 // Since we don't know which of the variants is present for this 

308 // CHOICE field, we'll parse this as a generic TLV encoded with 

309 // EXPLICIT, so `read_explicit_element` will consume the EXPLICIT 

310 // wrapper tag, and the TLV data will contain the variant. 

311 let tlv = parser.read_explicit_element::<asn1::Tlv<'_>>(*n)?; 

312 let type_without_explicit = AnnotatedType { 

313 inner: ann_type.inner.clone_ref(py), 

314 annotation: pyo3::Py::new( 

315 py, 

316 Annotation { 

317 default: None, 

318 encoding: None, 

319 size: None, 

320 }, 

321 )?, 

322 }; 

323 // Parse the TLV data (which contains the field without the EXPLICIT 

324 // wrapper) 

325 asn1::parse(tlv.full_data(), |d| { 

326 decode_annotated_type(py, d, &type_without_explicit) 

327 }) 

328 } 

329 } 

330}

332pub(crate) fn decode_annotated_type<'a>( 

333 py: pyo3::Python<'a>, 

334 parser: &mut Parser<'a>, 

335 ann_type: &AnnotatedType, 

336) -> ParseResult<pyo3::Bound<'a, pyo3::PyAny>> { 

337 let inner = ann_type.inner.get(); 

338 let annotation = &ann_type.annotation.get(); 

339 let encoding = &annotation.encoding; 

340  

341 // Handle DEFAULT annotation if field is not present (by 

342 // returning the default value) 

343 if let Some(default) = &ann_type.annotation.get().default { 

344 match parser.peek_tag() { 

345 Some(next_tag) if is_tag_valid_for_type(py, next_tag, inner, encoding) => (), 

346 _ => return Ok(default.bind(py).clone()), 

347 } 

348 } 

349  

350 let decoded = match &inner { 

351 Type::Sequence(cls, fields) => { 

352 let seq_parse_result = read_value::<asn1::Sequence<'_>>(parser, encoding)?; 

353  

354 seq_parse_result.parse(|d| -> ParseResult<pyo3::Bound<'a, pyo3::PyAny>> { 

355 let kwargs = pyo3::types::PyDict::new(py); 

356 let fields = fields.bind(py); 

357 for (name, ann_type) in fields.into_iter() { 

358 let ann_type = ann_type.cast::<AnnotatedType>()?; 

359 let value = decode_annotated_type(py, d, ann_type.get())?; 

360 kwargs.set_item(name, value)?; 

361 } 

362 let val = cls.call(py, (), Some(&kwargs))?.into_bound(py); 

363 Ok(val) 

364 })? 

365 } 

366 Type::SequenceOf(cls) => { 

367 let seqof_parse_result = read_value::<asn1::Sequence<'_>>(parser, encoding)?; 

368  

369 seqof_parse_result.parse(|d| -> ParseResult<pyo3::Bound<'a, pyo3::PyAny>> { 

370 let inner_ann_type = cls.get(); 

371 let list = pyo3::types::PyList::empty(py); 

372 while !d.is_empty() { 

373 let val = decode_annotated_type(py, d, inner_ann_type)?; 

374 list.append(val)?; 

375 } 

376 check_size_constraint(&annotation.size, || list.len(), "SEQUENCE OF")?; 

377 Ok(list.into_any()) 

378 })? 

379 } 

380 Type::Set(cls, fields) => { 

381 let set_parse_result = read_value::<asn1::Set<'_>>(parser, encoding)?; 

382  

383 set_parse_result.parse(|d| -> ParseResult<pyo3::Bound<'a, pyo3::PyAny>> { 

384 let kwargs = pyo3::types::PyDict::new(py); 

385 let fields = fields.bind(py); 

386 for (name, ann_type) in fields.into_iter() { 

387 let ann_type = ann_type.cast::<AnnotatedType>()?; 

388 let value = decode_annotated_type(py, d, ann_type.get())?; 

389 kwargs.set_item(name, value)?; 

390 } 

391 let val = cls.call(py, (), Some(&kwargs))?.into_bound(py); 

392 Ok(val) 

393 })? 

394 } 

395 Type::SetOf(cls) => { 

396 let setof_parse_result = read_value::<asn1::Set<'_>>(parser, encoding)?; 

397  

398 setof_parse_result.parse(|d| -> ParseResult<pyo3::Bound<'a, pyo3::PyAny>> { 

399 let inner_ann_type = cls.get(); 

400 let list = pyo3::types::PyList::empty(py); 

401 while !d.is_empty() { 

402 let val = decode_annotated_type(py, d, inner_ann_type)?; 

403 list.append(val)?; 

404 } 

405 check_size_constraint(&annotation.size, || list.len(), "SET OF")?; 

406 Ok(pyo3::Bound::new( 

407 py, 

408 SetOf { 

409 inner: list.unbind(), 

410 }, 

411 )? 

412 .into_any()) 

413 })? 

414 } 

415 Type::Option(cls) => { 

416 match parser.peek_tag() { 

417 Some(t) if is_tag_valid_for_type(py, t, cls.get().inner.get(), encoding) => { 

418 // For optional types, annotations will always be associated to the `Optional` type 

419 // i.e: `Annotated[Optional[T], annotation]`, as opposed to the inner `T` type. 

420 // Therefore, when decoding the inner type `T` we must pass the annotation of the `Optional` 

421 let inner_ann_type = AnnotatedType { 

422 inner: cls.get().inner.clone_ref(py), 

423 annotation: ann_type.annotation.clone_ref(py), 

424 }; 

425 decode_annotated_type(py, parser, &inner_ann_type)? 

426 } 

427 _ => pyo3::types::PyNone::get(py).to_owned().into_any(), 

428 } 

429 } 

430 Type::Choice(ts) => match encoding { 

431 Some(e) => decode_choice_with_encoding(py, parser, ann_type, e.get())?, 

432 None => { 

433 for t in ts.bind(py) { 

434 let variant = t.cast::<Variant>()?.get(); 

435 match parser.peek_tag() { 

436 Some(tag) if is_tag_valid_for_variant(py, tag, variant, encoding) => { 

437 let decoded_value = 

438 decode_annotated_type(py, parser, variant.ann_type.get())?; 

439 return match &variant.tag_name { 

440 Some(tag_name) => Ok(variant 

441 .python_class 

442 .call1(py, (decoded_value, tag_name))? 

443 .into_bound(py)), 

444 None => Ok(decoded_value), 

445 }; 

446 } 

447 _ => continue, 

448 } 

449 } 

450 Err(CryptographyError::Py( 

451 pyo3::exceptions::PyValueError::new_err( 

452 "could not find matching variant when parsing CHOICE field", 

453 ), 

454 ))? 

455 } 

456 }, 

457 Type::ValueSet(cls, inner_type, value_map) => { 

458 decode_value_set(py, parser, cls, inner_type.get(), value_map, annotation)? 

459 } 

460 Type::PyBool() => decode_pybool(py, parser, encoding)?.into_any(), 

461 Type::PyInt() => decode_pyint(py, parser, encoding)?.into_any(), 

462 Type::PyBytes() => decode_pybytes(py, parser, annotation)?.into_any(), 

463 Type::PyStr() => decode_pystr(py, parser, annotation)?.into_any(), 

464 Type::PrintableString() => decode_printable_string(py, parser, annotation)?.into_any(), 

465 Type::IA5String() => decode_ia5_string(py, parser, annotation)?.into_any(), 

466 Type::ObjectIdentifier() => decode_oid(py, parser, annotation)?.into_any(), 

467 Type::UtcTime() => decode_utc_time(py, parser, encoding)?.into_any(), 

468 Type::GeneralizedTime() => decode_generalized_time(py, parser, encoding)?.into_any(), 

469 Type::BitString() => decode_bitstring(py, parser, annotation)?.into_any(), 

470 Type::Tlv() => decode_tlv(py, parser, encoding)?.into_any(), 

471 Type::Null() => decode_null(py, parser, encoding)?.into_any(), 

472 Type::Certificate() => decode_certificate(py, parser, encoding)?.into_any(), 

473 Type::CertificateSigningRequest() => decode_csr(py, parser, encoding)?.into_any(), 

474 Type::CertificateRevocationList() => decode_crl(py, parser, encoding)?.into_any(), 

475 Type::Name() => decode_name(py, parser, encoding)?, 

476 }; 

477  

478 match &ann_type.annotation.get().default { 

479 Some(default) if decoded.eq(default.bind(py))? => Err(CryptographyError::Py( 

480 pyo3::exceptions::PyValueError::new_err( 

481 "invalid DER: DEFAULT value was explicitly encoded", 

482 ), 

483 )), 

484 _ => Ok(decoded), 

485 } 

486}
