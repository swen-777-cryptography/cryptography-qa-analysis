19/// Parse all sections in a PEM file and return the first matching section. 

20/// If no matching sections are found, return an error. 

21pub(crate) fn find_in_pem( 

22 data: &[u8], 

23 filter_fn: fn(&pem::Pem) -> bool, 

24 no_match_err: &'static str, 

25) -> Result<pem::Pem, CryptographyError> { 

26 let all_sections = pem::parse_many(data)?; 

27 if all_sections.is_empty() { 

28 return Err(CryptographyError::from(pem::PemError::MalformedFraming)); 

29 } 

30 all_sections.into_iter().find(filter_fn).ok_or_else(|| { 

31 CryptographyError::from(pyo3::exceptions::PyValueError::new_err(no_match_err)) 

32 }) 

33}

56pub(crate) fn encode_name_entry<'p>( 

57 py: pyo3::Python<'_>, 

58 ka: &'p cryptography_keepalive::KeepAlive<pyo3::pybacked::PyBackedBytes>, 

59 py_name_entry: &pyo3::Bound<'_, pyo3::PyAny>, 

60) -> CryptographyResult<AttributeTypeValue<'p>> { 

61 let attr_type = py_name_entry.getattr(pyo3::intern!(py, "_type"))?; 

62 let tag = attr_type 

63 .getattr(pyo3::intern!(py, "value"))? 

64 .extract::<u8>()?; 

65 let raw_value = py_name_entry.getattr(pyo3::intern!(py, "value"))?; 

66 let value = if attr_type.is(&types::ASN1_TYPE_BIT_STRING.get(py)?) { 

67 AttributeValue::AnyString(RawTlv::new( 

68 asn1::BitString::TAG, 

69 ka.add(raw_value.extract()?), 

70 )) 

71 } else if attr_type.is(&types::ASN1_TYPE_BMP_STRING.get(py)?) { 

72 AttributeValue::BmpString( 

73 asn1::BMPString::new( 

74 ka.add( 

75 raw_value 

76 .call_method1(pyo3::intern!(py, "encode"), ("utf_16_be",))? 

77 .extract()?, 

78 ), 

79 ) 

80 .unwrap(), 

81 ) 

82 } else if attr_type.is(&types::ASN1_TYPE_UNIVERSAL_STRING.get(py)?) { 

83 AttributeValue::UniversalString( 

84 asn1::UniversalString::new( 

85 ka.add( 

86 raw_value 

87 .call_method1(pyo3::intern!(py, "encode"), ("utf_32_be",))? 

88 .extract()?, 

89 ), 

90 ) 

91 .unwrap(), 

92 ) 

93 } else { 

94 AttributeValue::AnyString(RawTlv::new( 

95 asn1::Tag::from_bytes(&[tag])?.0, 

96 ka.add( 

97 raw_value 

98 .call_method1(pyo3::intern!(py, "encode"), ("utf8",))? 

99 .extract()?, 

100 ), 

101 )) 

102 }; 

103 let py_oid = py_name_entry.getattr(pyo3::intern!(py, "oid"))?;

107 type_id: oid,

146pub(crate) fn encode_general_name<'a>( 

147 py: pyo3::Python<'_>, 

148 ka_bytes: &'a cryptography_keepalive::KeepAlive<pyo3::pybacked::PyBackedBytes>, 

149 ka_str: &'a cryptography_keepalive::KeepAlive<pyo3::pybacked::PyBackedStr>, 

150 gn: &pyo3::Bound<'a, pyo3::PyAny>, 

151) -> Result<GeneralName<'a>, CryptographyError> { 

152 let gn_type = gn.get_type(); 

153 let gn_value = gn.getattr(pyo3::intern!(py, "value"))?; 

154  

155 if gn_type.is(&types::DNS_NAME.get(py)?) { 

156 Ok(GeneralName::DNSName(UnvalidatedIA5String( 

157 ka_str.add(gn_value.extract()?), 

158 ))) 

159 } else if gn_type.is(&types::RFC822_NAME.get(py)?) { 

160 Ok(GeneralName::RFC822Name(UnvalidatedIA5String( 

161 ka_str.add(gn_value.extract()?), 

162 ))) 

163 } else if gn_type.is(&types::DIRECTORY_NAME.get(py)?) { 

164 let name = encode_name(py, ka_bytes, &gn_value)?; 

165 Ok(GeneralName::DirectoryName(name)) 

166 } else if gn_type.is(&types::OTHER_NAME.get(py)?) { 

167 let py_oid = gn.getattr(pyo3::intern!(py, "type_id"))?; 

168 Ok(GeneralName::OtherName(OtherName { 

169 type_id: py_oid_to_oid(py_oid)?, 

170 value: asn1::parse_single(ka_bytes.add(gn_value.extract()?)).map_err(|e| { 

171 pyo3::exceptions::PyValueError::new_err(format!( 

172 "OtherName value must be valid DER: {e:?}" 

173 )) 

174 })?, 

175 })) 

176 } else if gn_type.is(&types::UNIFORM_RESOURCE_IDENTIFIER.get(py)?) { 

177 Ok(GeneralName::UniformResourceIdentifier( 

178 UnvalidatedIA5String(ka_str.add(gn_value.extract()?)), 

179 )) 

180 } else if gn_type.is(&types::IP_ADDRESS.get(py)?) { 

181 Ok(GeneralName::IPAddress(ka_bytes.add( 

182 gn.call_method0(pyo3::intern!(py, "_packed"))?.extract()?, 

183 ))) 

184 } else if gn_type.is(&types::REGISTERED_ID.get(py)?) { 

185 let oid = py_oid_to_oid(gn_value)?; 

186 Ok(GeneralName::RegisteredID(oid)) 

187 } else { 

188 Err(CryptographyError::from( 

189 pyo3::exceptions::PyValueError::new_err("Unsupported GeneralName type"), 

190 )) 

191 } 

192}

227fn parse_name_attribute<'p>( 

228 py: pyo3::Python<'p>, 

229 attribute: AttributeTypeValue<'_>, 

230) -> CryptographyResult<pyo3::Bound<'p, pyo3::PyAny>> { 

231 let oid = oid_to_py_oid(py, &attribute.type_id)?; 

232 let tag_val = attribute.value.tag().as_u8().ok_or_else(|| { 

233 CryptographyError::from(pyo3::exceptions::PyValueError::new_err( 

234 "Long-form tags are not supported in NameAttribute values", 

235 )) 

236 })?; 

237 // A value whose tag is not one of the recognised string types is not a 

238 // valid AttributeValue; surface it as a parse error rather than letting the 

239 // dict lookup raise a bare KeyError. 

240 let py_tag = types::ASN1_TYPE_TO_ENUM 

241 .get(py)? 

242 .get_item(tag_val) 

243 .map_err(|_| { 

244 asn1::ParseError::new(asn1::ParseErrorKind::UnexpectedTag { 

245 actual: attribute.value.tag(), 

246 }) 

247 })?; 

248 let py_data = match attribute.value { 

249 AttributeValue::AnyString(s) => { 

250 if s.tag() == asn1::BitString::TAG { 

251 pyo3::types::PyBytes::new(py, s.data()).into_any() 

252 } else { 

253 let parsed = std::str::from_utf8(s.data()) 

254 .map_err(|_| asn1::ParseError::new(asn1::ParseErrorKind::InvalidValue))?; 

255 pyo3::types::PyString::new(py, parsed).into_any() 

256 } 

257 } 

258 AttributeValue::PrintableString(printable_string) => { 

259 pyo3::types::PyString::new(py, printable_string.as_str()).into_any() 

260 } 

261 AttributeValue::UniversalString(universal_string) => { 

262 let py_bytes = pyo3::types::PyBytes::new(py, universal_string.as_utf32_be_bytes()); 

263 py_bytes.call_method1(pyo3::intern!(py, "decode"), ("utf_32_be",))? 

264 } 

265 AttributeValue::BmpString(bmp_string) => { 

266 let py_bytes = pyo3::types::PyBytes::new(py, bmp_string.as_utf16_be_bytes()); 

267 py_bytes.call_method1(pyo3::intern!(py, "decode"), ("utf_16_be",))? 

268 } 

269 }; 

270 let kwargs = [(pyo3::intern!(py, "_validate"), false)].into_py_dict(py)?; 

271 Ok(types::NAME_ATTRIBUTE 

272 .get(py)? 

273 .call((oid, py_data, py_tag), Some(&kwargs))?) 

274}

290pub(crate) fn parse_general_name<'p>( 

291 py: pyo3::Python<'p>, 

292 gn: GeneralName<'_>, 

293) -> CryptographyResult<pyo3::Bound<'p, pyo3::PyAny>> { 

294 let py_gn = match gn { 

295 GeneralName::OtherName(data) => { 

296 let oid = oid_to_py_oid(py, &data.type_id)?; 

297 types::OTHER_NAME 

298 .get(py)? 

299 .call1((oid, data.value.full_data()))? 

300 } 

301 GeneralName::RFC822Name(data) => types::RFC822_NAME 

302 .get(py)? 

303 .call_method1(pyo3::intern!(py, "_init_without_validation"), (data.0,))?, 

304 GeneralName::DNSName(data) => types::DNS_NAME 

305 .get(py)? 

306 .call_method1(pyo3::intern!(py, "_init_without_validation"), (data.0,))?, 

307 GeneralName::DirectoryName(data) => { 

308 let py_name = parse_name(py, data.unwrap_read())?; 

309 types::DIRECTORY_NAME.get(py)?.call1((py_name,))? 

310 } 

311 GeneralName::UniformResourceIdentifier(data) => types::UNIFORM_RESOURCE_IDENTIFIER 

312 .get(py)? 

313 .call_method1(pyo3::intern!(py, "_init_without_validation"), (data.0,))?, 

314 GeneralName::IPAddress(data) => { 

315 if data.len() == 4 || data.len() == 16 { 

316 let addr = types::IPADDRESS_IPADDRESS.get(py)?.call1((data,))?; 

317 types::IP_ADDRESS.get(py)?.call1((addr,))? 

318 } else { 

319 // if it's not an IPv4 or IPv6 we assume it's an IPNetwork and 

320 // verify length in this function. 

321 create_ip_network(py, data)? 

322 } 

323 } 

324 GeneralName::RegisteredID(data) => { 

325 let oid = oid_to_py_oid(py, &data)?; 

326 types::REGISTERED_ID.get(py)?.call1((oid,))? 

327 } 

328 _ => { 

329 return Err(CryptographyError::from( 

330 exceptions::UnsupportedGeneralNameType::new_err( 

331 "x400Address/EDIPartyName are not supported types", 

332 ), 

333 )) 

334 } 

335 }; 

336 Ok(py_gn) 

337}

351fn create_ip_network<'p>( 

352 py: pyo3::Python<'p>, 

353 data: &[u8], 

354) -> CryptographyResult<pyo3::Bound<'p, pyo3::PyAny>> { 

355 let prefix = match data.len() { 

356 8 => { 

357 let num = u32::from_be_bytes(data[4..].try_into().unwrap()); 

358 ipv4_netmask(num) 

359 } 

360 32 => { 

361 let num = u128::from_be_bytes(data[16..].try_into().unwrap()); 

362 ipv6_netmask(num) 

363 } 

364 _ => Err(CryptographyError::from(pyo3::exceptions::PyValueError::new_err( 

365 format!("Invalid IPNetwork, must be 8 bytes for IPv4 and 32 bytes for IPv6. Found length: {}", data.len()), 

366 ))), 

367 }; 

368 let base = types::IPADDRESS_IPADDRESS 

369 .get(py)? 

370 .call1((pyo3::types::PyBytes::new(py, &data[..data.len() / 2]),))?; 

371 let net = format!( 

372 "{}/{}", 

373 base.getattr(pyo3::intern!(py, "exploded"))? 

374 .extract::<pyo3::pybacked::PyBackedStr>()?, 

375 prefix? 

376 ); 

377 let addr = types::IPADDRESS_IPNETWORK.get(py)?.call1((net,))?; 

378 Ok(types::IP_ADDRESS.get(py)?.call1((addr,))?) 

379}

381fn ipv4_netmask(num: u32) -> Result<u32, CryptographyError> { 

382 if num.leading_ones() + num.trailing_zeros() != 32 { 

383 return Err(CryptographyError::from( 

384 pyo3::exceptions::PyValueError::new_err("Invalid netmask"), 

385 )); 

386 } 

387 Ok((!num).leading_zeros()) 

388}

399pub(crate) fn parse_and_cache_extensions< 

400 'p, 

401 F: Fn(&Extension<'p>) -> Result<Option<pyo3::Bound<'p, pyo3::PyAny>>, CryptographyError>, 

402>( 

403 py: pyo3::Python<'p>, 

404 cached_extensions: &pyo3::sync::PyOnceLock<pyo3::Py<pyo3::PyAny>>, 

405 raw_extensions: &Option<RawExtensions<'p>>, 

406 parse_ext: F, 

407) -> pyo3::PyResult<pyo3::Py<pyo3::PyAny>> { 

408 cached_extensions 

409 .get_or_try_init(py, || { 

410 let extensions = match Extensions::from_raw_extensions(raw_extensions.as_ref()) { 

411 Ok(extensions) => extensions, 

412 Err(DuplicateExtensionsError(oid)) => { 

413 let oid_obj = oid_to_py_oid(py, &oid)?; 

414 return Err(exceptions::DuplicateExtension::new_err(( 

415 format!("Duplicate {} extension found", oid), 

416 oid_obj.unbind(), 

417 ))); 

418 } 

419 }; 

420  

421 let exts = pyo3::types::PyList::empty(py); 

422 for raw_ext in extensions.iter() { 

423 let oid_obj = oid_to_py_oid(py, &raw_ext.extn_id)?; 

424  

425 let extn_value = match parse_ext(&raw_ext)? { 

426 Some(e) => e, 

427 None => types::UNRECOGNIZED_EXTENSION 

428 .get(py)? 

429 .call1((oid_obj.clone(), raw_ext.extn_value))?, 

430 }; 

431 let ext_obj = 

432 types::EXTENSION 

433 .get(py)? 

434 .call1((oid_obj, raw_ext.critical, extn_value))?; 

435 exts.append(ext_obj)?; 

436 } 

437 Ok(types::EXTENSIONS.get(py)?.call1((exts,))?.unbind()) 

438 }) 

439 .map(|p| p.clone_ref(py)) 

440}

442pub(crate) fn encode_extensions< 

443 'p, 

444 F: Fn( 

445 pyo3::Python<'_>, 

446 &asn1::ObjectIdentifier, 

447 &pyo3::Bound<'_, pyo3::PyAny>, 

448 ) -> CryptographyResult<Option<Vec<u8>>>, 

449>( 

450 py: pyo3::Python<'p>, 

451 ka_vec: &'p cryptography_keepalive::KeepAlive<Vec<u8>>, 

452 ka_bytes: &'p cryptography_keepalive::KeepAlive<pyo3::pybacked::PyBackedBytes>, 

453 py_exts: &pyo3::Bound<'p, pyo3::PyAny>, 

454 encode_ext: F, 

455) -> pyo3::PyResult<Option<RawExtensions<'p>>> { 

456 let mut exts = vec![]; 

457 for py_ext in py_exts.try_iter()? { 

458 let py_ext = py_ext?; 

459 let py_oid = py_ext.getattr(pyo3::intern!(py, "oid"))?; 

460 let oid = py_oid_to_oid(py_oid)?; 

461  

462 let ext_val = py_ext.getattr(pyo3::intern!(py, "value"))?; 

463 if ext_val.is_instance(&types::UNRECOGNIZED_EXTENSION.get(py)?)? { 

464 exts.push(Extension { 

465 extn_id: oid, 

466 critical: py_ext.getattr(pyo3::intern!(py, "critical"))?.extract()?, 

467 extn_value: ka_bytes.add(ext_val.getattr(pyo3::intern!(py, "value"))?.extract()?), 

468 }); 

469 continue; 

470 } 

471 match encode_ext(py, &oid, &ext_val)? { 

472 Some(data) => { 

473 exts.push(Extension { 

474 extn_id: oid, 

475 critical: py_ext.getattr(pyo3::intern!(py, "critical"))?.extract()?, 

476 extn_value: ka_vec.add(data), 

477 }); 

478 } 

479 None => { 

480 return Err(pyo3::exceptions::PyNotImplementedError::new_err(format!( 

481 "Extension not supported: {oid}" 

482 ))) 

483 } 

484 } 

485 } 

486 if exts.is_empty() { 

487 return Ok(None); 

488 } 

489 Ok(Some(Asn1ReadableOrWritable::new_write( 

490 asn1::SequenceOfWriter::new(exts), 

491 ))) 

492}

554// Convert Python's datetime objects to a tuple of `asn1::DateTime` and 

555// microseconds. 

556pub(crate) fn py_to_datetime_with_microseconds( 

557 py: pyo3::Python<'_>, 

558 val: pyo3::Bound<'_, pyo3::types::PyDateTime>, 

559) -> pyo3::PyResult<(asn1::DateTime, Option<u32>)> { 

560 // We treat naive datetimes as UTC times, while aware datetimes get 

561 // normalized to UTC before conversion. 

562 let val_utc = if val.get_tzinfo().is_none() { 

563 val.into_any() 

564 } else { 

565 let utc = pyo3::types::PyTzInfo::utc(py)?; 

566 val.call_method1(pyo3::intern!(py, "astimezone"), (utc,))? 

567 }; 

568  

569 let datetime = asn1::DateTime::new( 

570 val_utc.getattr(pyo3::intern!(py, "year"))?.extract()?, 

571 val_utc.getattr(pyo3::intern!(py, "month"))?.extract()?, 

572 val_utc.getattr(pyo3::intern!(py, "day"))?.extract()?, 

573 val_utc.getattr(pyo3::intern!(py, "hour"))?.extract()?, 

574 val_utc.getattr(pyo3::intern!(py, "minute"))?.extract()?, 

575 val_utc.getattr(pyo3::intern!(py, "second"))?.extract()?, 

576 ) 

577 .unwrap(); 

578  

579 let microseconds: u32 = val_utc 

580 .getattr(pyo3::intern!(py, "microsecond"))? 

581 .extract()?; 

582 let microseconds = if microseconds > 0 { 

583 Some(microseconds) 

584 } else { 

585 None 

586 }; 

587 Ok((datetime, microseconds)) 

588}
