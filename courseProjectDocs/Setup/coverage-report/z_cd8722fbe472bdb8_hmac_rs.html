25impl Hmac { 

26 // On BoringSSL and AWS-LC, the length is a size_t, so the length conversion is a 

27 // no-op. 

28 #[cfg_attr( 

29 any(CRYPTOGRAPHY_IS_BORINGSSL, CRYPTOGRAPHY_IS_AWSLC), 

30 allow(clippy::useless_conversion) 

31 )] 

32 pub fn new(key: &[u8], md: openssl::hash::MessageDigest) -> OpenSSLResult<Hmac> { 

33 // SAFETY: All FFI conditions are handled. 

34 unsafe { 

35 let h = Hmac::from_ptr(cvt_p(ffi::HMAC_CTX_new())?); 

36 cvt(ffi::HMAC_Init_ex( 

37 h.as_ptr(), 

38 key.as_ptr().cast(), 

39 key.len() 

40 .try_into() 

41 .expect("Key too long for OpenSSL's length type"), 

42 md.as_ptr(), 

43 ptr::null_mut(), 

44 ))?; 

45 Ok(h) 

46 } 

47 } 

48} 

49  

50impl HmacRef { 

51 pub fn update(&mut self, data: &[u8]) -> OpenSSLResult<()> { 

52 // SAFETY: All FFI conditions are handled. 

53 unsafe { 

54 cvt(ffi::HMAC_Update(self.as_ptr(), data.as_ptr(), data.len()))?; 

55 } 

56 Ok(()) 

57 } 

58  

59 pub fn finish(&mut self) -> OpenSSLResult<DigestBytes> { 

60 let mut buf = [0; ffi::EVP_MAX_MD_SIZE as usize]; 

61 let mut len = ffi::EVP_MAX_MD_SIZE as std::os::raw::c_uint; 

62 // SAFETY: All FFI conditions are handled. 

63 unsafe { 

64 cvt(ffi::HMAC_Final(self.as_ptr(), buf.as_mut_ptr(), &mut len))?; 

65 } 

66 Ok(DigestBytes { 

67 buf, 

68 len: len.try_into().unwrap(), 

69 }) 

70 } 

71  

72 pub fn copy(&self) -> OpenSSLResult<Hmac> { 

73 // SAFETY: All FFI conditions are handled. 

74 unsafe { 

75 let h = Hmac::from_ptr(cvt_p(ffi::HMAC_CTX_new())?); 

76 cvt(ffi::HMAC_CTX_copy(h.as_ptr(), self.as_ptr()))?; 

77 Ok(h) 

78 } 

79 } 

80}
