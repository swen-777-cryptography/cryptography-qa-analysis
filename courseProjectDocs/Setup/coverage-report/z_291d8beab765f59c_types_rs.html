13/// Represents a DNS name can be used in X.509 name matching. 

14/// 

15/// A `DNSName` is an `asn1::IA5String` with additional invariant preservations 

16/// per [RFC 5280 4.2.1.6], which in turn uses the preferred name syntax defined 

17/// in [RFC 1034 3.5] and amended in [RFC 1123 2.1]. 

18/// 

19/// Non-ASCII domain names (i.e., internationalized names) must be pre-encoded; 

20/// comparisons are case-insensitive. 

21/// 

22/// [RFC 5280 4.2.1.6]: https://datatracker.ietf.org/doc/html/rfc5280#section-4.2.1.6 

23/// [RFC 1034 3.5]: https://datatracker.ietf.org/doc/html/rfc1034#section-3.5 

24/// [RFC 1123 2.1]: https://datatracker.ietf.org/doc/html/rfc1123#section-2.1 

25/// 

26/// ```rust 

27/// # use cryptography_x509_verification::types::DNSName; 

28/// assert_eq!(DNSName::new("foo.com").unwrap(), DNSName::new("FOO.com").unwrap()); 

29/// ``` 

30#[derive(Clone, Debug)] 

31pub struct DNSName<'a>(asn1::IA5String<'a>); 

32  

33impl<'a> DNSName<'a> { 

34 pub fn new(value: &'a str) -> Option<Self> { 

35 // Domains cannot be empty and must (practically) 

36 // be less than 253 characters (255 in RFC 1034's octet encoding). 

37 if value.is_empty() || value.len() > 253 { 

38 None 

39 } else { 

40 for label in value.split('.') { 

41 // Individual labels cannot be empty; cannot exceed 63 characters; 

42 // cannot start or end with `-`. 

43 // NOTE: RFC 1034's grammar prohibits consecutive hyphens, but these 

44 // are used as part of the IDN prefix (e.g. `xn--`)'; we allow them here. 

45 if label.is_empty() 

46 || label.len() > 63 

47 || label.starts_with('-') 

48 || label.ends_with('-') 

49 { 

50 return None; 

51 } 

52  

53 // Labels must only contain `a-zA-Z0-9-`. 

54 if !label.chars().all(|c| c.is_ascii_alphanumeric() || c == '-') { 

55 return None; 

56 } 

57 } 

58 asn1::IA5String::new(value).map(Self) 

59 } 

60 } 

61  

62 pub fn as_str(&self) -> &'a str { 

63 self.0.as_str() 

64 } 

65  

66 /// Return this `DNSName`'s parent domain, if it has one. 

67 /// 

68 /// ```rust 

69 /// # use cryptography_x509_verification::types::DNSName; 

70 /// let domain = DNSName::new("foo.example.com").unwrap(); 

71 /// assert_eq!(domain.parent().unwrap().as_str(), "example.com"); 

72 /// ``` 

73 pub fn parent(&self) -> Option<Self> { 

74 match self.as_str().split_once('.') { 

75 Some((_, parent)) => Self::new(parent), 

76 None => None, 

77 } 

78 } 

79  

80 /// Returns this DNS name's labels, in reversed order 

81 /// (from top-level domain to most-specific subdomain). 

82 fn rlabels(&self) -> impl Iterator<Item = &'_ str> { 

83 self.as_str().rsplit('.') 

84 } 

85  

86 /// Returns true if this domain is a subdomain of the other domain. 

87 fn is_subdomain_of(&self, other: &DNSName<'_>) -> bool { 

88 // NOTE: This is nearly identical to `DNSConstraint::matches`, 

89 // except that the subdomain must be strictly longer than the parent domain. 

90 self.as_str().len() > other.as_str().len() 

91 && self 

92 .rlabels() 

93 .zip(other.rlabels()) 

94 .all(|(a, o)| a.eq_ignore_ascii_case(o)) 

95 } 

96} 

97  

98impl PartialEq for DNSName<'_> { 

99 fn eq(&self, other: &Self) -> bool { 

100 // DNS names are always case-insensitive. 

101 self.as_str().eq_ignore_ascii_case(other.as_str()) 

102 } 

103}

105/// Represents either a DNS name or a DNS wildcard for use in X.509 name 

106/// matching. 

107/// 

108/// A `DNSPattern` represents a subset of the domain name wildcard matching 

109/// behavior defined in [RFC 6125 6.4.3]. In particular, all DNS patterns 

110/// must either be exact matches (post-normalization) *or* a single wildcard 

111/// matching a full label in the left-most label position. Partial label matching 

112/// (e.g. `f*o.example.com`) is not supported, nor is non-left-most matching 

113/// (e.g. `foo.*.example.com`). 

114/// 

115/// [RFC 6125 6.4.3]: https://datatracker.ietf.org/doc/html/rfc6125#section-6.4.3 

116#[derive(Debug, PartialEq)] 

117pub enum DNSPattern<'a> { 

118 Exact(DNSName<'a>), 

119 Wildcard(DNSName<'a>), 

120} 

121  

122impl<'a> DNSPattern<'a> { 

123 pub fn new(pat: &'a str) -> Option<Self> { 

124 if let Some(pat) = pat.strip_prefix("*.") { 

125 DNSName::new(pat).map(Self::Wildcard) 

126 } else { 

127 DNSName::new(pat).map(Self::Exact) 

128 } 

129 } 

130  

131 pub fn matches(&self, name: &DNSName<'_>) -> bool { 

132 match self { 

133 Self::Exact(pat) => pat == name, 

134 Self::Wildcard(pat) => match name.parent() { 

135 Some(ref parent) => pat == parent, 

136 // No parent means we have a single label; wildcards cannot match single labels. 

137 None => false, 

138 }, 

139 } 

140 } 

141}

143/// A `DNSConstraint` represents a DNS name constraint as defined in [RFC 5280 4.2.1.10]. 

144/// 

145/// [RFC 5280 4.2.1.10]: https://datatracker.ietf.org/doc/html/rfc5280#section-4.2.1.10 

146pub struct DNSConstraint<'a>(DNSName<'a>); 

147  

148impl<'a> DNSConstraint<'a> { 

149 pub fn new(pattern: &'a str) -> Option<Self> { 

150 DNSName::new(pattern).map(Self) 

151 } 

152  

153 /// Returns true if the given exact `DNSName` falls within this 

154 /// constraint's subtree. 

155 /// 

156 /// Per RFC 5280, a name satisfies the constraint if it can be constructed 

157 /// by adding zero or more labels to the left-hand side of the constraint's 

158 /// name (i.e. it is the constraint's name, or a subdomain of it). 

159 fn contains(&self, name: &DNSName<'_>) -> bool { 

160 // NOTE: This may seem like an obtuse way to perform label matching, 

161 // but it saves us a few allocations: doing a substring check instead 

162 // would require us to clone each string and do case normalization. 

163 // Note also that we check the length in advance: Rust's zip 

164 // implementation terminates with the shorter iterator, so we need 

165 // to first check that the candidate name is at least as long as 

166 // the constraint it's matching against. 

167 name.as_str().len() >= self.0.as_str().len() 

168 && self 

169 .0 

170 .rlabels() 

171 .zip(name.rlabels()) 

172 .all(|(a, o)| a.eq_ignore_ascii_case(o)) 

173 } 

174  

175 /// Returns true if the given `DNSPattern` is permitted by this constraint, 

176 /// for use with a `permittedSubtrees` name constraint. 

177 /// 

178 /// A pattern is permitted only if *every* name it can represent falls 

179 /// within the constraint's subtree. An exact name is permitted by ordinary 

180 /// subtree containment (per RFC 5280). 

181 /// 

182 /// Wildcard patterns are not covered by RFC 5280; we define their behavior 

183 /// here. A wildcard pattern `*.X` is permitted only if its base name `X` 

184 /// itself falls within the constraint's subtree. This is stricter than 

185 /// mere overlap: `*.example.com` is *not* permitted by `foo.example.com`, 

186 /// since it can also expand to a sibling such as `bar.example.com` that 

187 /// lies outside the permitted subtree. 

188 pub fn permits(&self, pattern: &DNSPattern<'_>) -> bool { 

189 match pattern { 

190 DNSPattern::Exact(name) => self.contains(name), 

191 DNSPattern::Wildcard(base) => self.contains(base), 

192 } 

193 } 

194  

195 /// Returns true if the given `DNSPattern` is excluded by this constraint, 

196 /// for use with an `excludedSubtrees` name constraint. 

197 /// 

198 /// A pattern is excluded if *any* name it can represent falls within the 

199 /// constraint's subtree. An exact name is excluded by ordinary subtree 

200 /// containment (per RFC 5280). 

201 /// 

202 /// Wildcard patterns are not covered by RFC 5280; we define their behavior 

203 /// here. A wildcard pattern `*.X` is excluded if it overlaps the subtree 

204 /// at all, which happens in two subtly distinct cases: 

205 /// 

206 /// 1. The constraint is more specific than the wildcard, e.g. constraint 

207 /// `bar.example.com` and pattern `*.example.com` (which can expand to 

208 /// `bar.example.com`). This is handled by `DNSPattern::matches`. 

209 /// 2. The wildcard's base name falls within the subtree, e.g. constraint 

210 /// `example.com` and pattern `*.example.com`. This is handled by 

211 /// `DNSConstraint::contains`. 

212 pub fn excludes(&self, pattern: &DNSPattern<'_>) -> bool { 

213 match pattern { 

214 DNSPattern::Exact(name) => self.contains(name), 

215 DNSPattern::Wildcard(base) => pattern.matches(&self.0) || self.contains(base), 

216 } 

217 } 

218}

223/// An `IPAddress` represents an IP address as defined in [RFC 5280 4.2.1.6]. 

224/// 

225/// [RFC 5280 4.2.1.6]: https://datatracker.ietf.org/doc/html/rfc5280#section-4.2.1.6 

226impl IPAddress { 

227 #[allow(clippy::should_implement_trait)] 

228 pub fn from_str(s: &str) -> Option<Self> { 

229 IpAddr::from_str(s).ok().map(Self::from) 

230 } 

231  

232 /// Constructs an `IPAddress` from a slice. The provided data must be 

233 /// 4 (IPv4) or 16 (IPv6) bytes in "network byte order", as specified by 

234 /// [RFC 5280]. 

235 /// 

236 /// [RFC 5280]: https://datatracker.ietf.org/doc/html/rfc5280#section-4.2.1.6 

237 pub fn from_bytes(b: &[u8]) -> Option<Self> { 

238 match b.len() { 

239 4 => { 

240 let b: [u8; 4] = b.try_into().ok()?; 

241 Some(IpAddr::from(b).into()) 

242 } 

243 16 => { 

244 let b: [u8; 16] = b.try_into().ok()?; 

245 Some(IpAddr::from(b).into()) 

246 } 

247 _ => None, 

248 } 

249 } 

250  

251 /// Parses the octets of the `IPAddress` as a mask. If it is well-formed, 

252 /// i.e., has only one contiguous block of set bits starting from the most 

253 /// significant bit, a prefix is returned. 

254 pub fn as_prefix(&self) -> Option<u8> { 

255 let (leading, total) = match self.0 { 

256 IpAddr::V4(a) => { 

257 let data = u32::from_be_bytes(a.octets()); 

258 (data.leading_ones(), data.count_ones()) 

259 } 

260 IpAddr::V6(a) => { 

261 let data = u128::from_be_bytes(a.octets()); 

262 (data.leading_ones(), data.count_ones()) 

263 } 

264 }; 

265  

266 if leading != total { 

267 None 

268 } else { 

269 Some(leading as u8) 

270 } 

271 } 

272  

273 /// Returns a new `IPAddress` with the first `prefix` bits of the `IPAddress`. 

274 /// 

275 /// ```rust 

276 /// # use cryptography_x509_verification::types::IPAddress; 

277 /// let ip = IPAddress::from_str("192.0.2.1").unwrap(); 

278 /// assert_eq!(ip.mask(24), IPAddress::from_str("192.0.2.0").unwrap()); 

279 /// ``` 

280 pub fn mask(&self, prefix: u8) -> Self { 

281 match self.0 { 

282 IpAddr::V4(a) => { 

283 let prefix = 32u8.saturating_sub(prefix).into(); 

284 let masked = u32::from_be_bytes(a.octets()) 

285 & u32::MAX 

286 .checked_shr(prefix) 

287 .unwrap_or(0) 

288 .checked_shl(prefix) 

289 .unwrap_or(0); 

290 Self::from_bytes(&masked.to_be_bytes()).unwrap() 

291 } 

292 IpAddr::V6(a) => { 

293 let prefix = 128u8.saturating_sub(prefix).into(); 

294 let masked = u128::from_be_bytes(a.octets()) 

295 & u128::MAX 

296 .checked_shr(prefix) 

297 .unwrap_or(0) 

298 .checked_shl(prefix) 

299 .unwrap_or(0); 

300 Self::from_bytes(&masked.to_be_bytes()).unwrap() 

301 } 

302 } 

303 } 

304} 

305  

306impl From<IpAddr> for IPAddress { 

307 fn from(addr: IpAddr) -> Self { 

308 Self(addr) 

309 } 

310}

312#[derive(Debug, PartialEq, Eq)] 

313pub struct IPConstraint { 

314 address: IPAddress, 

315 prefix: u8, 

316} 

317  

318/// An `IPConstraint` represents a CIDR-style IP address range used in a name constraints 

319/// extension, as defined by [RFC 5280 4.2.1.10]. 

320/// 

321/// [RFC 5280 4.2.1.10]: https://datatracker.ietf.org/doc/html/rfc5280#section-4.2.1.10 

322impl IPConstraint { 

323 /// Constructs an `IPConstraint` from a slice. The input slice must be 8 (IPv4) 

324 /// or 32 (IPv6) bytes long and contain two IP addresses, the first being 

325 /// a subnet and the second defining the subnet's mask. 

326 /// 

327 /// The subnet mask must contain only one contiguous run of set bits starting 

328 /// from the most significant bit. For example, a valid IPv4 subnet mask would 

329 /// be FF FF 00 00, whereas an invalid IPv4 subnet mask would be FF EF 00 00. 

330 pub fn from_bytes(b: &[u8]) -> Option<Self> { 

331 let slice_idx = match b.len() { 

332 8 => 4, 

333 32 => 16, 

334 _ => return None, 

335 }; 

336  

337 let prefix = IPAddress::from_bytes(&b[slice_idx..])?.as_prefix()?; 

338 Some(IPConstraint { 

339 address: IPAddress::from_bytes(&b[..slice_idx])?.mask(prefix), 

340 prefix, 

341 }) 

342 } 

343  

344 /// Determines if the `addr` is within the `IPConstraint`. 

345 /// 

346 /// ```rust 

347 /// # use cryptography_x509_verification::types::{IPAddress, IPConstraint}; 

348 /// let range_bytes = b"\xc6\x33\x64\x00\xff\xff\xff\x00"; 

349 /// let range = IPConstraint::from_bytes(range_bytes).unwrap(); 

350 /// assert!(range.matches(&IPAddress::from_str("198.51.100.42").unwrap())); 

351 /// ``` 

352 pub fn matches(&self, addr: &IPAddress) -> bool { 

353 self.address == addr.mask(self.prefix) 

354 } 

355}

357/// An `RFC822Name` represents an email address, as defined in [RFC 822 6.1] 

358/// and as amended by [RFC 2821 4.1.2]. In particular, it represents the `Mailbox` 

359/// rule from RFC 2821's grammar. 

360/// 

361/// This type does not currently support the quoted local-part form; email 

362/// addresses that use this form will be rejected. 

363/// 

364/// [RFC 822 6.1]: https://datatracker.ietf.org/doc/html/rfc822#section-6.1 

365/// [RFC 2821 4.1.2]: https://datatracker.ietf.org/doc/html/rfc2821#section-4.1.2 

366#[derive(PartialEq)] 

367pub struct RFC822Name<'a> { 

368 pub mailbox: IA5String<'a>, 

369 pub domain: DNSName<'a>, 

370} 

371  

372impl<'a> RFC822Name<'a> { 

373 pub fn new(value: &'a str) -> Option<Self> { 

374 // Mailbox = Local-part "@" Domain 

375 // Both must be present. 

376 let (local_part, domain) = value.split_once('@')?; 

377 let local_part = IA5String::new(local_part)?; 

378  

379 // Local-part = Dot-string / Quoted-string 

380 // NOTE(ww): We do not support the Quoted-string form, for now. 

381 // 

382 // Dot-string: Atom *("." Atom) 

383 // Atom = 1*atext 

384 // 

385 // NOTE(ww): `atext`'s production is in RFC 2822 3.2.4. 

386 for component in local_part.as_str().split('.') { 

387 if component.is_empty() 

388 || !component 

389 .chars() 

390 .all(|c| c.is_ascii_alphanumeric() || ATEXT_CHARS.contains(c)) 

391 { 

392 return None; 

393 } 

394 } 

395  

396 Some(Self { 

397 mailbox: local_part, 

398 domain: DNSName::new(domain)?, 

399 }) 

400 } 

401} 

402  

403/// An `RFC822Constraint` represents a Name Constraint on email addresses. 

404pub enum RFC822Constraint<'a> { 

405 /// A constraint for an exact match on a specific email address. 

406 Exact(RFC822Name<'a>), 

407 /// A constraint for any mailbox on a particular domain. 

408 OnDomain(DNSName<'a>), 

409 /// A constraint for any mailbox *within* a particular domain. 

410 /// For example, `InDomain("example.com")` will match `foo@bar.example.com` 

411 /// but not `foo@example.com`, since `bar.example.com` is in `example.com` 

412 /// but `example.com` is not within itself. 

413 InDomain(DNSName<'a>), 

414} 

415  

416impl<'a> RFC822Constraint<'a> { 

417 pub fn new(constraint: &'a str) -> Option<Self> { 

418 if let Some(constraint) = constraint.strip_prefix('.') { 

419 Some(Self::InDomain(DNSName::new(constraint)?)) 

420 } else if let Some(email) = RFC822Name::new(constraint) { 

421 Some(Self::Exact(email)) 

422 } else { 

423 Some(Self::OnDomain(DNSName::new(constraint)?)) 

424 } 

425 } 

426  

427 pub fn matches(&self, email: &RFC822Name<'_>) -> bool { 

428 match self { 

429 Self::Exact(pat) => pat == email, 

430 Self::OnDomain(pat) => &email.domain == pat, 

431 Self::InDomain(pat) => email.domain.is_subdomain_of(pat), 

432 } 

433 } 

434}

641 #[test] 

642 fn test_dnsconstraint_permits_wildcard() { 

643 let com = DNSConstraint::new("com").unwrap(); 

644 let example_com = DNSConstraint::new("example.com").unwrap(); 

645 let foo_example_com = DNSConstraint::new("foo.example.com").unwrap(); 

646 let any_example_com = DNSPattern::new("*.example.com").unwrap(); 

647  

648 // A wildcard `*.example.com` is permitted only by constraints whose 

649 // subtree contains *every* name the wildcard can expand to, i.e. those 

650 // that contain `example.com` itself. 

651 assert!(com.permits(&any_example_com)); 

652 assert!(example_com.permits(&any_example_com)); 

653  

654 // A constraint more specific than the wildcard's base does *not* 

655 // permit it: the wildcard can expand to siblings outside the subtree 

656 // (e.g. `*.example.com` can be `bar.example.com`, which lies outside 

657 // `foo.example.com`). 

658 assert!(!foo_example_com.permits(&any_example_com)); 

659 }

661 #[test] 

662 fn test_dnsconstraint_excludes_wildcard() { 

663 let com = DNSConstraint::new("com").unwrap(); 

664 let example_com = DNSConstraint::new("example.com").unwrap(); 

665 let bar_example_com = DNSConstraint::new("bar.example.com").unwrap(); 

666 let baz_bar_example_com = DNSConstraint::new("baz.bar.example.com").unwrap(); 

667 let any_example_com = DNSPattern::new("*.example.com").unwrap(); 

668  

669 // A wildcard `*.example.com` is excluded by any constraint whose 

670 // subtree it overlaps, including constraints more specific than the 

671 // wildcard's base. 

672 assert!(com.excludes(&any_example_com)); 

673 assert!(example_com.excludes(&any_example_com)); 

674 assert!(bar_example_com.excludes(&any_example_com)); 

675  

676 // A constraint on `baz.bar.example.com` doesn't overlap 

677 // `*.example.com`, since `baz.bar.example.com` matches zero or more 

678 // sublabels of `baz.bar.example.com` while `*.example.com` matches 

679 // exactly one sublabel of `example.com`. 

680 assert!(!baz_bar_example_com.excludes(&any_example_com)); 

681 }

894 fn test_rfc822constraint_matches() { 

895 { 

896 let exact = RFC822Constraint::new("foo@example.com").unwrap(); 

897  

898 // Ordinary exact match. 

899 assert!(exact.matches(&RFC822Name::new("foo@example.com").unwrap())); 

900 // Case changes are okay in the domain. 

901 assert!(exact.matches(&RFC822Name::new("foo@EXAMPLE.com").unwrap())); 

902  

903 // Case changes are not okay in the mailbox. 

904 assert!(!exact.matches(&RFC822Name::new("Foo@example.com").unwrap())); 

905 assert!(!exact.matches(&RFC822Name::new("FOO@example.com").unwrap())); 

906  

907 // Different mailboxes and domains do not match. 

908 assert!(!exact.matches(&RFC822Name::new("foo.bar@example.com").unwrap())); 

909 assert!(!exact.matches(&RFC822Name::new("foo@sub.example.com").unwrap())); 

910 } 

911  

912 { 

913 let on_domain = RFC822Constraint::new("example.com").unwrap(); 

914  

915 // Ordinary domain matches. 

916 assert!(on_domain.matches(&RFC822Name::new("foo@example.com").unwrap())); 

917 assert!(on_domain.matches(&RFC822Name::new("bar@example.com").unwrap())); 

918 assert!(on_domain.matches(&RFC822Name::new("foo.bar@example.com").unwrap())); 

919 assert!(on_domain.matches(&RFC822Name::new("foo!bar@example.com").unwrap())); 

920 // Case changes are okay in the domain and in the mailbox, 

921 // since any mailbox on the domain is okay. 

922 assert!(on_domain.matches(&RFC822Name::new("foo@EXAMPLE.com").unwrap())); 

923 assert!(on_domain.matches(&RFC822Name::new("FOO@example.com").unwrap())); 

924  

925 // Subdomains and other domains do not match. 

926 assert!(!on_domain.matches(&RFC822Name::new("foo@sub.example.com").unwrap())); 

927 assert!(!on_domain.matches(&RFC822Name::new("foo@localhost").unwrap())); 

928 } 

929  

930 { 

931 let in_domain = RFC822Constraint::new(".example.com").unwrap(); 

932  

933 // Any subdomain and mailbox matches. 

934 assert!(in_domain.matches(&RFC822Name::new("foo@sub.example.com").unwrap())); 

935 assert!(in_domain.matches(&RFC822Name::new("foo@sub.sub.example.com").unwrap())); 

936 assert!(in_domain.matches(&RFC822Name::new("foo@com.example.example.com").unwrap())); 

937 assert!(in_domain.matches(&RFC822Name::new("foo.bar@com.example.example.com").unwrap())); 

938 assert!(in_domain.matches(&RFC822Name::new("foo!bar@com.example.example.com").unwrap())); 

939 assert!(in_domain.matches(&RFC822Name::new("bar@com.example.example.com").unwrap())); 

940 // Case changes are okay in the subdomains and in the mailbox, since any mailbox 

941 // in the domain is okay. 

942 assert!(in_domain.matches(&RFC822Name::new("foo@SUB.example.com").unwrap())); 

943 assert!(in_domain.matches(&RFC822Name::new("foo@sub.EXAMPLE.com").unwrap())); 

944 assert!(in_domain.matches(&RFC822Name::new("foo@sub.example.COM").unwrap())); 

945 assert!(in_domain.matches(&RFC822Name::new("FOO@sub.example.COM").unwrap())); 

946 assert!(in_domain.matches(&RFC822Name::new("FOO@sub.example.com").unwrap())); 

947  

948 // Superdomains and other domains do not match. 

949 assert!(!in_domain.matches(&RFC822Name::new("foo@example.com").unwrap())); 

950 assert!(!in_domain.matches(&RFC822Name::new("foo@com").unwrap()));
