90#[pyo3::pyfunction] 

91fn encrypt_and_serialize<'p>( 

92 py: pyo3::Python<'p>, 

93 builder: &pyo3::Bound<'p, pyo3::PyAny>, 

94 content_encryption_algorithm: &pyo3::Bound<'p, pyo3::PyAny>, 

95 encoding: crate::serialization::Encoding, 

96 options: &pyo3::Bound<'p, pyo3::types::PyList>, 

97) -> CryptographyResult<pyo3::Bound<'p, pyo3::types::PyBytes>> { 

98 let raw_data: CffiBuf<'p> = builder.getattr(pyo3::intern!(py, "_data"))?.extract()?; 

99 let text_mode = options.contains(types::PKCS7_TEXT.get(py)?)?; 

100 let data_with_header = if options.contains(types::PKCS7_BINARY.get(py)?)? { 

101 Cow::Borrowed(raw_data.as_bytes()) 

102 } else { 

103 smime_canonicalize(raw_data.as_bytes(), text_mode).0 

104 }; 

105  

106 // Get the content encryption algorithm 

107 let content_encryption_algorithm_type = content_encryption_algorithm; 

108 let key_size = content_encryption_algorithm_type.getattr(pyo3::intern!(py, "key_size"))?; 

109 let key = crate::backend::rand::get_rand_bytes(py, key_size.floor_div(8)?.extract()?)?; 

110 let content_encryption_algorithm = content_encryption_algorithm_type.call1((&key,))?; 

111  

112 // Get the mode 

113 let iv = crate::backend::rand::get_rand_bytes(py, 16)?; 

114 let cbc_mode = types::CBC.get(py)?.call1((&iv,))?; 

115  

116 let encrypted_content = symmetric_encrypt( 

117 py, 

118 content_encryption_algorithm, 

119 cbc_mode, 

120 &data_with_header, 

121 )?; 

122  

123 let py_recipients: Vec<pyo3::Bound<'p, x509::certificate::Certificate>> = builder 

124 .getattr(pyo3::intern!(py, "_recipients"))? 

125 .extract()?; 

126  

127 let mut recipient_infos = vec![]; 

128 let padding = types::PKCS1V15.get(py)?.call0()?; 

129 let ka_bytes = cryptography_keepalive::KeepAlive::new(); 

130 for cert in py_recipients.iter() { 

131 // Currently, keys are encrypted with RSA (PKCS #1 v1.5), which the S/MIME v3.2 RFC 

132 // specifies as MUST support (https://datatracker.ietf.org/doc/html/rfc5751#section-2.3) 

133 // however rest of S/MIME v3.2 is not currently supported 

134 let encrypted_key = cert 

135 .call_method0(pyo3::intern!(py, "public_key"))? 

136 .call_method1(pyo3::intern!(py, "encrypt"), (&key, &padding))? 

137 .extract::<pyo3::pybacked::PyBackedBytes>()?; 

138  

139 recipient_infos.push(pkcs7::RecipientInfo { 

140 version: 0, 

141 issuer_and_serial_number: pkcs7::IssuerAndSerialNumber { 

142 issuer: cert.get().raw.borrow_dependent().tbs_cert.issuer.clone(), 

143 serial_number: cert.get().raw.borrow_dependent().tbs_cert.serial, 

144 }, 

145 key_encryption_algorithm: AlgorithmIdentifier { 

146 oid: asn1::DefinedByMarker::marker(), 

147 params: AlgorithmParameters::Rsa(Some(())), 

148 }, 

149 encrypted_key: ka_bytes.add(encrypted_key), 

150 }); 

151 } 

152  

153 // Prepare the algorithm parameters 

154 let algorithm_parameters = if content_encryption_algorithm_type.eq(types::AES128.get(py)?)? { 

155 AlgorithmParameters::Aes128Cbc(iv.extract()?) 

156 } else { 

157 AlgorithmParameters::Aes256Cbc(iv.extract()?) 

158 }; 

159  

160 let enveloped_data = pkcs7::EnvelopedData { 

161 version: 0, 

162 recipient_infos: common::Asn1ReadableOrWritable::new_write(asn1::SetOfWriter::new( 

163 &recipient_infos, 

164 )), 

165  

166 encrypted_content_info: pkcs7::EncryptedContentInfo { 

167 content_type: PKCS7_DATA_OID, 

168 content_encryption_algorithm: AlgorithmIdentifier { 

169 oid: asn1::DefinedByMarker::marker(), 

170 params: algorithm_parameters, 

171 }, 

172 encrypted_content: Some(&encrypted_content), 

173 }, 

174 }; 

175  

176 let content_info = pkcs7::ContentInfo { 

177 _content_type: asn1::DefinedByMarker::marker(), 

178 content: pkcs7::Content::EnvelopedData(asn1::Explicit::new(Box::new(enveloped_data))), 

179 }; 

180 let ci_bytes = asn1::write_single(&content_info)?; 

181  

182 if encoding == crate::serialization::Encoding::SMIME { 

183 Ok(types::SMIME_ENVELOPED_ENCODE 

184 .get(py)? 

185 .call1((&*ci_bytes,))? 

186 .extract()?) 

187 } else { 

188 // Handles the DER, PEM, and error cases 

189 encode_der_data(py, "PKCS7".to_string(), ci_bytes, encoding) 

190 } 

191}

210 py: pyo3::Python<'p>, 

211 data: &[u8],

233#[pyo3::pyfunction] 

234fn decrypt_der<'p>( 

235 py: pyo3::Python<'p>, 

236 data: &[u8], 

237 certificate: pyo3::Bound<'p, x509::certificate::Certificate>, 

238 private_key: pyo3::Bound<'p, pyo3::types::PyAny>, 

239 options: &pyo3::Bound<'p, pyo3::types::PyList>, 

240) -> CryptographyResult<pyo3::Bound<'p, pyo3::types::PyBytes>> { 

241 // Check the decrypt parameters 

242 check_decrypt_parameters(py, &certificate, &private_key, options)?; 

243  

244 // Decrypt the data 

245 let content_info = asn1::parse_single::<pkcs7::ContentInfo<'_>>(data)?; 

246 let plain_content = match content_info.content { 

247 pkcs7::Content::EnvelopedData(data) => { 

248 // Extract enveloped data 

249 let enveloped_data = data.into_inner(); 

250  

251 // Get recipients, and the one matching with the given certificate (if any) 

252 let mut recipient_infos = enveloped_data.recipient_infos.unwrap_read().clone(); 

253 let recipient_certificate = certificate.get().raw.borrow_dependent(); 

254 let recipient_serial_number = recipient_certificate.tbs_cert.serial; 

255 let recipient_issuer = recipient_certificate.tbs_cert.issuer.clone(); 

256 let found_recipient_info = recipient_infos.find(|info| { 

257 info.issuer_and_serial_number.serial_number == recipient_serial_number 

258 && info.issuer_and_serial_number.issuer == recipient_issuer 

259 }); 

260  

261 // Raise error when no recipient is found 

262 let recipient_info = match found_recipient_info { 

263 Some(info) => info, 

264 None => { 

265 return Err(CryptographyError::from( 

266 pyo3::exceptions::PyValueError::new_err( 

267 "No recipient found that matches the given certificate.", 

268 ), 

269 )); 

270 } 

271 }; 

272  

273 // The function can decrypt content encrypted with AES-128-CBC, which the S/MIME v3.2 

274 // RFC specifies as MUST support, and AES-256-CBC, which is specified as SHOULD+ 

275 // support. More info: https://datatracker.ietf.org/doc/html/rfc5751#section-2.7 

276 // however rest of S/MIME v3.2 is not currently supported 

277 let algorithm_identifier = enveloped_data 

278 .encrypted_content_info 

279 .content_encryption_algorithm; 

280 let (algorithm_type, key_size, iv) = match algorithm_identifier.params { 

281 AlgorithmParameters::Aes128Cbc(iv) => (&types::AES128, 16, iv), 

282 AlgorithmParameters::Aes256Cbc(iv) => (&types::AES256, 32, iv), 

283 _ => { 

284 return Err(CryptographyError::from( 

285 exceptions::UnsupportedAlgorithm::new_err(( 

286 "Only AES (with key sizes 128 or 256) with CBC mode is currently supported for content decryption.", 

287 exceptions::Reasons::UNSUPPORTED_SERIALIZATION, 

288 )), 

289 )); 

290 } 

291 }; 

292  

293 // Raise error when the key encryption algorithm is not RSA 

294 if recipient_info.key_encryption_algorithm.oid() != &oid::RSA_OID { 

295 return Err(CryptographyError::from( 

296 exceptions::UnsupportedAlgorithm::new_err(( 

297 "Only RSA with PKCS #1 v1.5 padding is currently supported for key decryption.", 

298 exceptions::Reasons::UNSUPPORTED_SERIALIZATION, 

299 )), 

300 )); 

301 } 

302  

303 // Unwrap the content encryption key. RFC 3218 requires that a 

304 // failure to recover a usable key is not distinguishable from 

305 // recovering a key that simply isn't the right one -- otherwise 

306 // this becomes a Bleichenbacher oracle for anything that decrypts 

307 // attacker-supplied data. We substitute a random key of the 

308 // expected size, so that every failure surfaces as the same 

309 // content decryption error. 

310 let padding = types::PKCS1V15.get(py)?.call0()?; 

311 let random_key = crate::backend::rand::get_rand_bytes(py, key_size)?; 

312 let key = match private_key.call_method1( 

313 pyo3::intern!(py, "decrypt"), 

314 (recipient_info.encrypted_key, &padding), 

315 ) { 

316 Ok(key) => { 

317 let key = key.extract::<pyo3::Bound<'_, pyo3::types::PyBytes>>()?; 

318 if key.as_bytes().len() == key_size { 

319 key 

320 } else { 

321 random_key 

322 } 

323 } 

324 // A ValueError here is entirely attacker controlled (invalid 

325 // padding, or an encrypted key of the wrong length) and must 

326 // stay silent. Any other exception reflects a caller mistake 

327 // rather than the contents of the message. 

328 Err(e) if e.is_instance_of::<pyo3::exceptions::PyValueError>(py) => random_key, 

329 Err(e) => return Err(e.into()), 

330 }; 

331  

332 let algorithm = algorithm_type.get(py)?.call1((key,))?; 

333 let mode = types::CBC 

334 .get(py)? 

335 .call1((pyo3::types::PyBytes::new(py, &iv),))?; 

336  

337 // Decrypt the content using the key and proper algorithm 

338 let encrypted_content = match enveloped_data.encrypted_content_info.encrypted_content { 

339 Some(content) => content, 

340 None => { 

341 return Err(CryptographyError::from( 

342 pyo3::exceptions::PyValueError::new_err( 

343 "The EnvelopedData structure does not contain encrypted content.", 

344 ), 

345 )); 

346 } 

347 }; 

348 let decrypted_content = symmetric_decrypt(py, algorithm, mode, encrypted_content)?; 

349 pyo3::types::PyBytes::new(py, decrypted_content.as_slice()) 

350 } 

351 _ => { 

352 return Err(CryptographyError::from( 

353 pyo3::exceptions::PyValueError::new_err( 

354 "The PKCS7 data is not an EnvelopedData structure.", 

355 ), 

356 )); 

357 } 

358 }; 

359  

360 // If text_mode, remove the headers after checking the content type 

361 let plain_data = if options.contains(types::PKCS7_TEXT.get(py)?)? { 

362 let stripped_data = types::SMIME_REMOVE_TEXT_HEADERS 

363 .get(py)? 

364 .call1((plain_content.as_bytes(),))?; 

365 pyo3::types::PyBytes::new(py, stripped_data.extract()?) 

366 } else { 

367 pyo3::types::PyBytes::new(py, plain_content.as_bytes()) 

368 }; 

369  

370 Ok(plain_data) 

371}

373fn check_decrypt_parameters<'p>( 

374 py: pyo3::Python<'p>, 

375 certificate: &pyo3::Bound<'p, x509::certificate::Certificate>, 

376 private_key: &pyo3::Bound<'p, pyo3::PyAny>, 

377 options: &pyo3::Bound<'p, pyo3::types::PyList>, 

378) -> Result<(), CryptographyError> { 

379 // Check if RSA encryption with PKCS1 v1.5 padding is supported (dependent of FIPS mode) 

380 if cryptography_openssl::fips::is_enabled() { 

381 return Err(CryptographyError::from( 

382 exceptions::UnsupportedAlgorithm::new_err(( 

383 "RSA with PKCS1 v1.5 padding is not supported by this version of OpenSSL.", 

384 exceptions::Reasons::UNSUPPORTED_PADDING, 

385 )), 

386 )); 

387 } 

388  

389 // Check if all options are from the PKCS7Options enum 

390 let pkcs7_options = types::PKCS7_OPTIONS.get(py)?; 

391 for opt in options.iter() { 

392 if !opt.is_instance(&pkcs7_options)? { 

393 return Err(CryptographyError::from( 

394 pyo3::exceptions::PyValueError::new_err( 

395 "options must be from the PKCS7Options enum", 

396 ), 

397 )); 

398 } 

399 } 

400  

401 // Check if any option is not PKCS7Options::Text 

402 let text_option = types::PKCS7_TEXT.get(py)?; 

403 for opt in options.iter() { 

404 if !opt.eq(text_option.clone())? { 

405 return Err(CryptographyError::from( 

406 pyo3::exceptions::PyValueError::new_err( 

407 "Only the following options are supported for decryption: Text", 

408 ), 

409 )); 

410 } 

411 } 

412  

413 // Check if certificate's public key is an RSA public key 

414 let public_key_type = types::RSA_PUBLIC_KEY.get(py)?; 

415 if !certificate 

416 .call_method0(pyo3::intern!(py, "public_key"))? 

417 .is_instance(&public_key_type)? 

418 { 

419 return Err(CryptographyError::from( 

420 pyo3::exceptions::PyTypeError::new_err( 

421 "Only certificate with RSA public keys are supported at this time.", 

422 ), 

423 )); 

424 } 

425  

426 // Check if private_key is an instance of RSA private key 

427 let private_key_type = types::RSA_PRIVATE_KEY.get(py)?; 

428 if !private_key.is_instance(&private_key_type)? { 

429 return Err(CryptographyError::from( 

430 pyo3::exceptions::PyTypeError::new_err( 

431 "Only RSA private keys are supported at this time.", 

432 ), 

433 )); 

434 } 

435  

436 Ok(()) 

437}

439pub(crate) fn symmetric_decrypt( 

440 py: pyo3::Python<'_>, 

441 algorithm: pyo3::Bound<'_, pyo3::PyAny>, 

442 mode: pyo3::Bound<'_, pyo3::PyAny>, 

443 data: &[u8], 

444) -> CryptographyResult<Vec<u8>> { 

445 let block_size = algorithm 

446 .getattr(pyo3::intern!(py, "block_size"))? 

447 .extract()?; 

448  

449 let mut cipher = 

450 ciphers::CipherContext::new(py, algorithm, mode, openssl::symm::Mode::Decrypt)?; 

451  

452 // Decrypt the data 

453 let mut decrypted_data = vec![0; data.len() + (block_size / 8)]; 

454 let count = cipher.update_into(py, data, &mut decrypted_data)?; 

455 let final_block = cipher.finalize(py)?; 

456 assert!(final_block.as_bytes().is_empty()); 

457 decrypted_data.truncate(count); 

458  

459 // Unpad the data 

460 let mut unpadder = PKCS7UnpaddingContext::new(block_size); 

461 let unpadded_first_blocks = unpadder.update(py, CffiBuf::from_bytes(py, &decrypted_data))?; 

462 let unpadded_last_block = unpadder.finalize(py)?; 

463  

464 let unpadded_data = [ 

465 unpadded_first_blocks.as_bytes(), 

466 unpadded_last_block.as_bytes(), 

467 ] 

468 .concat(); 

469  

470 Ok(unpadded_data) 

471}

473#[pyo3::pyfunction] 

474fn sign_and_serialize<'p>( 

475 py: pyo3::Python<'p>, 

476 builder: &pyo3::Bound<'p, pyo3::PyAny>, 

477 encoding: crate::serialization::Encoding, 

478 options: &pyo3::Bound<'p, pyo3::types::PyList>, 

479) -> CryptographyResult<pyo3::Bound<'p, pyo3::types::PyBytes>> { 

480 let raw_data: CffiBuf<'p> = builder.getattr(pyo3::intern!(py, "_data"))?.extract()?; 

481 let text_mode = options.contains(types::PKCS7_TEXT.get(py)?)?; 

482 let (data_with_header, data_without_header) = 

483 if options.contains(types::PKCS7_BINARY.get(py)?)? { 

484 ( 

485 Cow::Borrowed(raw_data.as_bytes()), 

486 Cow::Borrowed(raw_data.as_bytes()), 

487 ) 

488 } else { 

489 smime_canonicalize(raw_data.as_bytes(), text_mode) 

490 }; 

491  

492 let content_type_bytes = asn1::write_single(&pkcs7::PKCS7_DATA_OID)?; 

493 let now = x509::common::datetime_now(py)?; 

494 let signing_time_bytes = asn1::write_single(&x509::certificate::time_from_datetime(now)?)?; 

495 let smime_cap_bytes = asn1::write_single(&asn1::SequenceOfWriter::new([ 

496 // Subset of values OpenSSL provides: 

497 // https://github.com/openssl/openssl/blob/667a8501f0b6e5705fd611d5bb3ca24848b07154/crypto/pkcs7/pk7_smime.c#L150 

498 // removing all the ones that are bad cryptography 

499 asn1::SequenceOfWriter::new([oid::AES_256_CBC_OID]), 

500 asn1::SequenceOfWriter::new([oid::AES_192_CBC_OID]), 

501 asn1::SequenceOfWriter::new([oid::AES_128_CBC_OID]), 

502 ]))?; 

503  

504 #[allow(clippy::type_complexity)] 

505 let py_signers: Vec<( 

506 pyo3::PyRef<'p, x509::certificate::Certificate>, 

507 pyo3::Bound<'_, pyo3::PyAny>, 

508 pyo3::Bound<'_, pyo3::PyAny>, 

509 pyo3::Bound<'_, pyo3::PyAny>, 

510 )> = builder.getattr(pyo3::intern!(py, "_signers"))?.extract()?; 

511  

512 let py_certs: Vec<pyo3::PyRef<'p, x509::certificate::Certificate>> = builder 

513 .getattr(pyo3::intern!(py, "_additional_certs"))? 

514 .extract()?; 

515  

516 let mut signer_infos = vec![]; 

517 let mut digest_algs = vec![]; 

518 let mut certs = py_certs 

519 .iter() 

520 .map(|p| p.raw.borrow_dependent().clone()) 

521 .collect::<Vec<_>>(); 

522  

523 let ka_vec = cryptography_keepalive::KeepAlive::new(); 

524 let ka_bytes = cryptography_keepalive::KeepAlive::new(); 

525 for (cert, py_private_key, py_hash_alg, rsa_padding) in py_signers.iter() { 

526 let (authenticated_attrs, signature) = 

527 if options.contains(&types::PKCS7_NO_ATTRIBUTES.get(py)?)? { 

528 ( 

529 None, 

530 x509::sign::sign_data( 

531 py, 

532 py_private_key.clone(), 

533 py_hash_alg.clone(), 

534 rsa_padding.clone(), 

535 None, 

536 &data_with_header, 

537 )?, 

538 ) 

539 } else { 

540 let mut authenticated_attrs = vec![ 

541 Attribute { 

542 type_id: PKCS7_CONTENT_TYPE_OID, 

543 values: common::Asn1ReadableOrWritable::new_write(asn1::SetOfWriter::new( 

544 [asn1::parse_single(&content_type_bytes).unwrap()], 

545 )), 

546 }, 

547 Attribute { 

548 type_id: PKCS7_SIGNING_TIME_OID, 

549 values: common::Asn1ReadableOrWritable::new_write(asn1::SetOfWriter::new( 

550 [asn1::parse_single(&signing_time_bytes).unwrap()], 

551 )), 

552 }, 

553 ]; 

554  

555 let digest = x509::ocsp::hash_data(py, py_hash_alg, &data_with_header)?; 

556 let digest_wrapped = ka_vec.add(asn1::write_single(&digest.as_bytes())?); 

557 authenticated_attrs.push(Attribute { 

558 type_id: PKCS7_MESSAGE_DIGEST_OID, 

559 values: common::Asn1ReadableOrWritable::new_write(asn1::SetOfWriter::new([ 

560 asn1::parse_single(digest_wrapped).unwrap(), 

561 ])), 

562 }); 

563  

564 if !options.contains(types::PKCS7_NO_CAPABILITIES.get(py)?)? { 

565 authenticated_attrs.push(Attribute { 

566 type_id: PKCS7_SMIME_CAP_OID, 

567 values: common::Asn1ReadableOrWritable::new_write(asn1::SetOfWriter::new( 

568 [asn1::parse_single(&smime_cap_bytes).unwrap()], 

569 )), 

570 }); 

571 } 

572  

573 let signed_data = 

574 asn1::write_single(&asn1::SetOfWriter::new(authenticated_attrs.as_slice()))?; 

575  

576 ( 

577 Some(common::Asn1ReadableOrWritable::new_write( 

578 asn1::SetOfWriter::new(authenticated_attrs), 

579 )), 

580 x509::sign::sign_data( 

581 py, 

582 py_private_key.clone(), 

583 py_hash_alg.clone(), 

584 rsa_padding.clone(), 

585 None, 

586 &signed_data, 

587 )?, 

588 ) 

589 }; 

590  

591 let digest_alg = x509::ocsp::HASH_NAME_TO_ALGORITHM_IDENTIFIERS[&*py_hash_alg 

592 .getattr(pyo3::intern!(py, "name"))? 

593 .extract::<pyo3::pybacked::PyBackedStr>()?] 

594 .clone(); 

595 // Technically O(n^2), but no one will have that many signers. 

596 if !digest_algs.contains(&digest_alg) { 

597 digest_algs.push(digest_alg.clone()); 

598 } 

599 certs.push(cert.raw.borrow_dependent().clone()); 

600  

601 signer_infos.push(pkcs7::SignerInfo { 

602 version: 1, 

603 issuer_and_serial_number: pkcs7::IssuerAndSerialNumber { 

604 issuer: cert.raw.borrow_dependent().tbs_cert.issuer.clone(), 

605 serial_number: cert.raw.borrow_dependent().tbs_cert.serial, 

606 }, 

607 digest_algorithm: digest_alg, 

608 authenticated_attributes: authenticated_attrs, 

609 digest_encryption_algorithm: compute_pkcs7_signature_algorithm( 

610 py, 

611 py_private_key.clone(), 

612 py_hash_alg.clone(), 

613 rsa_padding.clone(), 

614 )?, 

615 encrypted_digest: ka_bytes.add(signature), 

616 unauthenticated_attributes: None, 

617 }); 

618 } 

619  

620 let data_tlv_bytes; 

621 let content = if options.contains(types::PKCS7_DETACHED_SIGNATURE.get(py)?)? { 

622 None 

623 } else { 

624 data_tlv_bytes = asn1::write_single(&data_with_header.deref())?; 

625 Some(asn1::parse_single(&data_tlv_bytes).unwrap()) 

626 }; 

627  

628 let signed_data = pkcs7::SignedData { 

629 version: 1, 

630 digest_algorithms: common::Asn1ReadableOrWritable::new_write(asn1::SetOfWriter::new( 

631 &digest_algs, 

632 )), 

633 content_info: pkcs7::ContentInfo { 

634 _content_type: asn1::DefinedByMarker::marker(), 

635 content: pkcs7::Content::Data(content.map(asn1::Explicit::new)), 

636 }, 

637 certificates: if options.contains(types::PKCS7_NO_CERTS.get(py)?)? { 

638 None 

639 } else { 

640 Some(common::Asn1ReadableOrWritable::new_write( 

641 asn1::SetOfWriter::new(&certs), 

642 )) 

643 }, 

644 crls: None, 

645 signer_infos: common::Asn1ReadableOrWritable::new_write(asn1::SetOfWriter::new( 

646 &signer_infos, 

647 )), 

648 }; 

649  

650 let content_info = pkcs7::ContentInfo { 

651 _content_type: asn1::DefinedByMarker::marker(), 

652 content: pkcs7::Content::SignedData(asn1::Explicit::new(Box::new(signed_data))), 

653 }; 

654 let ci_bytes = asn1::write_single(&content_info)?; 

655  

656 if encoding == crate::serialization::Encoding::SMIME { 

657 let mic_algs = digest_algs 

658 .iter() 

659 .map(|d| OIDS_TO_MIC_NAME[&d.oid()]) 

660 .collect::<Vec<_>>() 

661 .join(","); 

662 Ok(types::SMIME_SIGNED_ENCODE 

663 .get(py)? 

664 .call1((&*data_without_header, &*ci_bytes, mic_algs, text_mode))? 

665 .extract()?) 

666 } else { 

667 // Handles the DER, PEM, and error cases 

668 encode_der_data(py, "PKCS7".to_string(), ci_bytes, encoding) 

669 } 

670}

672fn compute_pkcs7_signature_algorithm<'p>( 

673 py: pyo3::Python<'p>, 

674 private_key: pyo3::Bound<'p, pyo3::PyAny>, 

675 hash_algorithm: pyo3::Bound<'p, pyo3::PyAny>, 

676 rsa_padding: pyo3::Bound<'p, pyo3::PyAny>, 

677) -> pyo3::PyResult<common::AlgorithmIdentifier<'static>> { 

678 let key_type = x509::sign::identify_key_type(py, private_key.clone())?; 

679 let has_pss_padding = rsa_padding.is_instance(&types::PSS.get(py)?)?; 

680 // For RSA signatures (with no PSS padding), the OID is always the same no matter the 

681 // digest algorithm. See RFC 3370 (section 3.2). 

682 if key_type == x509::sign::KeyType::Rsa && !has_pss_padding { 

683 Ok(common::AlgorithmIdentifier { 

684 oid: asn1::DefinedByMarker::marker(), 

685 params: common::AlgorithmParameters::Rsa(Some(())), 

686 }) 

687 } else { 

688 x509::sign::compute_signature_algorithm(py, private_key, hash_algorithm, rsa_padding) 

689 } 

690}

692fn smime_canonicalize(data: &[u8], text_mode: bool) -> (Cow<'_, [u8]>, Cow<'_, [u8]>) { 

693 let mut new_data_with_header = vec![]; 

694 let mut new_data_without_header = vec![]; 

695 if text_mode { 

696 new_data_with_header.extend_from_slice(b"Content-Type: text/plain\r\n\r\n"); 

697 } 

698  

699 let mut last_idx = 0; 

700 for (i, c) in data.iter().copied().enumerate() { 

701 if c == b'\n' && (i == 0 || data[i - 1] != b'\r') { 

702 new_data_with_header.extend_from_slice(&data[last_idx..i]); 

703 new_data_with_header.push(b'\r'); 

704 new_data_with_header.push(b'\n'); 

705  

706 new_data_without_header.extend_from_slice(&data[last_idx..i]); 

707 new_data_without_header.push(b'\r'); 

708 new_data_without_header.push(b'\n'); 

709 last_idx = i + 1; 

710 } 

711 } 

712 // If there's stuff in new_data, that means we need to copy the rest of 

713 // data over. 

714 if !new_data_with_header.is_empty() { 

715 new_data_with_header.extend_from_slice(&data[last_idx..]); 

716 new_data_without_header.extend_from_slice(&data[last_idx..]); 

717 ( 

718 Cow::Owned(new_data_with_header), 

719 Cow::Owned(new_data_without_header), 

720 ) 

721 } else { 

722 (Cow::Borrowed(data), Cow::Borrowed(data)) 

723 } 

724}

726#[cfg(not(any(CRYPTOGRAPHY_IS_BORINGSSL, CRYPTOGRAPHY_IS_AWSLC)))] 

727fn load_pkcs7_certificates( 

728 py: pyo3::Python<'_>, 

729 pkcs7: Pkcs7, 

730) -> CryptographyResult<pyo3::Bound<'_, pyo3::types::PyList>> { 

731 let nid = pkcs7.type_().map(|t| t.nid()); 

732 if nid != Some(openssl::nid::Nid::PKCS7_SIGNED) { 

733 let nid_string = nid.map_or("empty".to_string(), |n| n.as_raw().to_string()); 

734 return Err(CryptographyError::from( 

735 exceptions::UnsupportedAlgorithm::new_err(( 

736 format!("Only basic signed structures are currently supported. NID for this data was {nid_string}"), 

737 exceptions::Reasons::UNSUPPORTED_SERIALIZATION, 

738 )), 

739 )); 

740 } 

741  

742 let signed_certificates = pkcs7.signed().and_then(|x| x.certificates()); 

743 match signed_certificates { 

744 None => Err(CryptographyError::from( 

745 pyo3::exceptions::PyValueError::new_err( 

746 "The provided PKCS7 has no certificate data, but a cert loading method was called.", 

747 ), 

748 )), 

749 Some(certificates) => { 

750 let result = pyo3::types::PyList::empty(py); 

751 for c in certificates { 

752 let cert_der = pyo3::types::PyBytes::new(py, c.to_der()?.as_slice()).unbind(); 

753 let cert = load_der_x509_certificate(py, cert_der, None)?; 

754 result.append(cert)?; 

755 } 

756 Ok(result) 

757 } 

758 } 

759}

761pub fn try_list_of_certificates<'p, F>( 

762 py: pyo3::Python<'p>, 

763 data: pyo3::Py<pyo3::types::PyBytes>, 

764 f: F, 

765) -> CryptographyResult<pyo3::Bound<'p, pyo3::types::PyList>> 

766where 

767 F: for<'a> FnOnce( 

768 &'a pyo3::Py<pyo3::types::PyBytes>, 

769 &mut dyn FnMut(RawCertificate<'a>) -> CryptographyResult<()>, 

770 ) -> CryptographyResult<()>, 

771{ 

772 let result = pyo3::types::PyList::empty(py); 

773 let mut cb = |val| { 

774 // SAFETY: based on the type of `F`, we know `val` must be derived from 

775 // data, and we know that `data.clone_ref(py)` makes any pointers into 

776 // the original one also valid. 

777 let raw_cert = certificate::OwnedCertificate::new(data.clone_ref(py), |_| unsafe { 

778 mem::transmute(val) 

779 }); 

780 result.append(pyo3::Bound::new( 

781 py, 

782 x509::certificate::Certificate { 

783 raw: raw_cert, 

784 cached_extensions: pyo3::sync::PyOnceLock::new(), 

785 cached_issuer: pyo3::sync::PyOnceLock::new(), 

786 cached_subject: pyo3::sync::PyOnceLock::new(), 

787 cached_public_key: pyo3::sync::PyOnceLock::new(), 

788 }, 

789 )?)?; 

790  

791 Ok(()) 

792 }; 

793 f(&data, &mut cb)?; 

794  

795 Ok(result) 

796}

798fn load_pkcs7_certificates_rust( 

799 py: pyo3::Python<'_>, 

800 data: pyo3::Py<pyo3::types::PyBytes>, 

801) -> CryptographyResult<pyo3::Bound<'_, pyo3::types::PyList>> { 

802 try_list_of_certificates(py, data, |data, cb| { 

803 let p7 = asn1::parse_single::<pkcs7::ContentInfo<'_>>(data.as_bytes(py))?; 

804 let pkcs7::Content::SignedData(signed_data) = p7.content else { 

805 return Err(CryptographyError::from( 

806 exceptions::UnsupportedAlgorithm::new_err(( 

807 "Only basic signed structures are currently supported.", 

808 exceptions::Reasons::UNSUPPORTED_SERIALIZATION, 

809 )), 

810 )); 

811 }; 

812 let Some(certs) = signed_data.into_inner().certificates else { 

813 return Err(CryptographyError::from( 

814 pyo3::exceptions::PyValueError::new_err( 

815 "The provided PKCS7 has no certificate data, but a cert loading method was called.", 

816 ), 

817 )); 

818 }; 

819 for c in certs.unwrap_read().clone() { 

820 cb(c)?; 

821 } 

822  

823 Ok(()) 

824 }) 

825}

828fn load_pem_pkcs7_certificates( 

829 py: pyo3::Python<'_>, 

830 data: pyo3::Py<pyo3::types::PyBytes>, 

831) -> CryptographyResult<pyo3::Bound<'_, pyo3::types::PyList>> { 

832 let pem_block = pem::parse(data.as_bytes(py))?; 

833 if pem_block.tag() != "PKCS7" { 

834 return Err(CryptographyError::from( 

835 pyo3::exceptions::PyValueError::new_err( 

836 "The provided PEM data does not have the PKCS7 tag.", 

837 ), 

838 )); 

839 } 

840 let data = pyo3::types::PyBytes::new(py, pem_block.contents()).unbind(); 

841  

842 load_der_pkcs7_certificates(py, data) 

843}

845#[pyo3::pyfunction] 

846fn load_der_pkcs7_certificates( 

847 py: pyo3::Python<'_>, 

848 data: pyo3::Py<pyo3::types::PyBytes>, 

849) -> CryptographyResult<pyo3::Bound<'_, pyo3::types::PyList>> { 

850 cfg_if::cfg_if! { 

851 if #[cfg(not(any(CRYPTOGRAPHY_IS_BORINGSSL, CRYPTOGRAPHY_IS_AWSLC)))] { 

852 let pkcs7_decoded = openssl::pkcs7::Pkcs7::from_der(data.as_bytes(py)).map_err(|_| { 

853 CryptographyError::from(pyo3::exceptions::PyValueError::new_err( 

854 "Unable to parse PKCS7 data", 

855 )) 

856 })?; 

857 let result = load_pkcs7_certificates(py, pkcs7_decoded)?; 

858 if let Err(e) = load_pkcs7_certificates_rust(py, data) { 

859 let err = pyo3::PyErr::from(e); 

860 let warning_cls = pyo3::exceptions::PyUserWarning::type_object(py); 

861 let message = CString::new(format!("PKCS#7 certificates could not be parsed as DER, falling back to parsing as BER. In the future, this may become an exception. Error details: {err}")).unwrap(); 

862 pyo3::PyErr::warn(py, &warning_cls, &message, 1)?; 

863 } 

864  

865 Ok(result) 

866 } else { 

867 load_pkcs7_certificates_rust(py, data) 

868 } 

869 }
