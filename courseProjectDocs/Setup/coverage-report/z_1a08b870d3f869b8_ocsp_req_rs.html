119 #[getter] 

120 fn extensions(&self, py: pyo3::Python<'_>) -> pyo3::PyResult<pyo3::Py<pyo3::PyAny>> { 

121 let tbs_request = &self.raw.borrow_dependent().tbs_request; 

122  

123 x509::parse_and_cache_extensions( 

124 py, 

125 &self.cached_extensions, 

126 &tbs_request.raw_request_extensions, 

127 |ext| { 

128 match ext.extn_id { 

129 oid::NONCE_OID => { 

130 // This is a disaster. RFC 2560 says that the contents of the nonce is 

131 // just the raw extension value. This is nonsense, since they're always 

132 // supposed to be ASN.1 TLVs. RFC 6960 correctly specifies that the 

133 // nonce is an OCTET STRING, and so you should unwrap the TLV to get 

134 // the nonce. So we try parsing as a TLV and fall back to just using 

135 // the raw value. 

136 let nonce = ext.value::<&[u8]>().unwrap_or(ext.extn_value); 

137 Ok(Some(types::OCSP_NONCE.get(py)?.call1((nonce,))?)) 

138 } 

139 oid::ACCEPTABLE_RESPONSES_OID => { 

140 let oids = ext.value::<asn1::SequenceOf<'_, asn1::ObjectIdentifier>>()?; 

141 let py_oids = pyo3::types::PyList::empty(py); 

142 for oid in oids { 

143 py_oids.append(oid_to_py_oid(py, &oid)?)?; 

144 } 

145  

146 Ok(Some( 

147 types::OCSP_ACCEPTABLE_RESPONSES 

148 .get(py)? 

149 .call1((py_oids,))?, 

150 )) 

151 } 

152 _ => Ok(None), 

153 } 

154 }, 

155 ) 

156 }
