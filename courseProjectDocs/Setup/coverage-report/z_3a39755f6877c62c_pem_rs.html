9/// Decrypts PEM encryption (legacy format with Proc-Type and DEK-Info headers). 

10/// Returns a tuple of (decrypted contents, was_encrypted flag). 

11/// 

12/// If no `Proc-Type` header is preseent, the PEM contents is returned 

13/// undecrypted. 

14/// 

15/// Supported ciphers: AES-128-CBC, AES-256-CBC, DES-EDE3-CBC 

16pub fn decrypt_pem<'a>( 

17 pem: &'a pem::Pem, 

18 password: Option<&[u8]>, 

19) -> KeyParsingResult<(Cow<'a, [u8]>, bool)> { 

20 match pem.headers().get("Proc-Type") { 

21 Some("4,ENCRYPTED") => { 

22 let dek_info = pem 

23 .headers() 

24 .get("DEK-Info") 

25 .ok_or(KeyParsingError::PemMissingDekInfo)?; 

26  

27 let (cipher_algorithm, iv) = dek_info 

28 .split_once(',') 

29 .ok_or(KeyParsingError::PemInvalidDekInfo)?; 

30  

31 let password = match password { 

32 None | Some(b"") => return Err(KeyParsingError::EncryptedKeyWithoutPassword), 

33 Some(p) => p, 

34 }; 

35  

36 // There's no RFC that defines these, but these are the ones in 

37 // very wide use that we support. 

38 let cipher = match cipher_algorithm { 

39 "AES-128-CBC" => openssl::symm::Cipher::aes_128_cbc(), 

40 "AES-256-CBC" => openssl::symm::Cipher::aes_256_cbc(), 

41 "DES-EDE3-CBC" => openssl::symm::Cipher::des_ede3_cbc(), 

42 _ => return Err(KeyParsingError::PemUnsupportedCipher), 

43 }; 

44  

45 let iv = cryptography_crypto::encoding::hex_decode(iv) 

46 .ok_or(KeyParsingError::PemInvalidIv)?; 

47  

48 let key = cryptography_crypto::pbkdf1::openssl_kdf( 

49 openssl::hash::MessageDigest::md5(), 

50 password, 

51 iv.get(..8) 

52 .ok_or(KeyParsingError::PemInvalidIv)? 

53 .try_into() 

54 .unwrap(), 

55 cipher.key_len(), 

56 ) 

57 .map_err(|_| KeyParsingError::PemUnableToDeriveKey)?; 

58  

59 let decrypted = openssl::symm::decrypt(cipher, &key, Some(&iv), pem.contents()) 

60 .map_err(|_| KeyParsingError::IncorrectPassword)?; 

61  

62 Ok((Cow::Owned(decrypted), true)) 

63 } 

64 Some(_) => Err(KeyParsingError::PemInvalidProcType), 

65 None => Ok((Cow::Borrowed(pem.contents()), false)), 

66 } 

67}

69/// PEM encoding configuration using LF line endings (Unix-style) 

70pub const ENCODE_CONFIG: pem::EncodeConfig = 

71 pem::EncodeConfig::new().set_line_ending(pem::LineEnding::LF); 

72  

73/// Encrypts DER data with legacy PEM encryption (Proc-Type and DEK-Info 

74/// headers). Returns PEM-formatted bytes with encryption headers. 

75/// 

76/// If password is empty, returns unencrypted PEM. Otherwise, encrypts using 

77/// AES-256-CBC. 

78pub fn encrypt_pem( 

79 tag: &str, 

80 der_data: &[u8], 

81 password: &[u8], 

82) -> crate::KeySerializationResult<Vec<u8>> { 

83 if password.is_empty() { 

84 // No encryption - just encode as PEM 

85 let pem = pem::Pem::new(tag, der_data); 

86 return Ok(pem::encode_config(&pem, ENCODE_CONFIG).into_bytes()); 

87 } 

88  

89 let cipher = openssl::symm::Cipher::aes_256_cbc(); 

90 let iv_len = cipher.iv_len().unwrap(); 

91 let mut iv = vec![0u8; iv_len]; 

92 cryptography_openssl::rand::rand_bytes(&mut iv)?; 

93  

94 // Derive key using MD5-based KDF (for compatibility with traditional 

95 // OpenSSL format) 

96 let key = cryptography_crypto::pbkdf1::openssl_kdf( 

97 openssl::hash::MessageDigest::md5(), 

98 password, 

99 iv.get(..8).unwrap().try_into().unwrap(), 

100 cipher.key_len(), 

101 )?; 

102  

103 // Encrypt the DER data 

104 let encrypted = openssl::symm::encrypt(cipher, &key, Some(&iv), der_data)?; 

105  

106 let iv_hex = cryptography_crypto::encoding::hex_encode(&iv); 

107  

108 let mut pem = pem::Pem::new(tag, encrypted); 

109 pem.headers_mut().add("Proc-Type", "4,ENCRYPTED").unwrap(); 

110 pem.headers_mut() 

111 .add("DEK-Info", &format!("AES-256-CBC,{iv_hex}")) 

112 .unwrap(); 

113  

114 Ok(pem::encode_config(&pem, ENCODE_CONFIG).into_bytes()) 

115}
