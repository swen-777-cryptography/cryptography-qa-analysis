24/// An IA5String ASN.1 element whose contents is not validated as meeting the 

25/// requirements (ASCII characters only), and instead is only known to be 

26/// valid UTF-8. 

27pub struct UnvalidatedIA5String<'a>(pub &'a str); 

28  

29impl<'a> asn1::SimpleAsn1Readable<'a> for UnvalidatedIA5String<'a> { 

30 const TAG: asn1::Tag = asn1::IA5String::TAG; 

31 fn parse_data(data: &'a [u8]) -> asn1::ParseResult<Self> { 

32 Ok(UnvalidatedIA5String(std::str::from_utf8(data).map_err( 

33 |_| asn1::ParseError::new(asn1::ParseErrorKind::InvalidValue), 

34 )?)) 

35 } 

36}

38impl asn1::SimpleAsn1Writable for UnvalidatedIA5String<'_> { 

39 type Error = asn1::WriteError; 

40 const TAG: asn1::Tag = asn1::IA5String::TAG; 

41 fn write_data(&self, dest: &mut asn1::WriteBuf) -> asn1::WriteResult { 

42 dest.push_slice(self.0.as_bytes())
