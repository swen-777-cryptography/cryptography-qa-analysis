186 &self, 

187 py: pyo3::Python<'p>, 

188 ) -> CryptographyResult<pyo3::Bound<'p, pyo3::types::PyBytes>> { 

189 let val = self.raw.borrow_dependent(); 

190 let mut tbs_precert = val.tbs_cert.clone(); 

191 // Remove the SCT list extension 

192 match val.extensions() { 

193 Ok(extensions) => { 

194 let ext_count = extensions 

195 .as_raw() 

196 .as_ref() 

197 .map_or(0, |raw| raw.unwrap_read().len()); 

198 let filtered_extensions: Vec<Extension<'_>> = extensions 

199 .iter() 

200 .filter(|x| x.extn_id != oid::PRECERT_SIGNED_CERTIFICATE_TIMESTAMPS_OID) 

201 .collect(); 

202 if filtered_extensions.len() == ext_count { 

203 return Err(CryptographyError::from( 

204 pyo3::exceptions::PyValueError::new_err( 

205 "Could not find pre-certificate SCT list extension", 

206 ), 

207 )); 

208 } 

209 let filtered_extensions: RawExtensions<'_> = Asn1ReadableOrWritable::new_write( 

210 asn1::SequenceOfWriter::new(filtered_extensions), 

211 ); 

212  

213 tbs_precert.raw_extensions = Some(filtered_extensions); 

214 let result = asn1::write_single(&tbs_precert)?; 

215 Ok(pyo3::types::PyBytes::new(py, &result)) 

216 } 

217 Err(DuplicateExtensionsError(oid)) => { 

218 let oid_obj = oid_to_py_oid(py, &oid)?; 

219 Err(exceptions::DuplicateExtension::new_err((

344 )) 

345 } 

346 _ => parse_cert_ext(py, ext), 

347 },

351 fn verify_directly_issued_by( 

352 &self, 

353 issuer: pyo3::PyRef<'_, Certificate>, 

354 ) -> CryptographyResult<()> { 

355 if self.raw.borrow_dependent().tbs_cert.signature_alg 

356 != self.raw.borrow_dependent().signature_alg 

357 { 

358 return Err(CryptographyError::from(pyo3::exceptions::PyValueError::new_err( 

359 "Inner and outer signature algorithms do not match. This is an invalid certificate." 

360 ))); 

361 }; 

362 if self.raw.borrow_dependent().tbs_cert.issuer 

363 != issuer.raw.borrow_dependent().tbs_cert.subject 

364 { 

365 return Err(CryptographyError::from( 

366 pyo3::exceptions::PyValueError::new_err( 

367 "Issuer certificate subject does not match certificate issuer.", 

368 ), 

369 )); 

370 }; 

371  

372 let ops = PyCryptoOps {}; 

373 let issuer_key = ops.public_key(issuer.raw.borrow_dependent())?; 

374 ops.verify_signed_by(self.raw.borrow_dependent(), &issuer_key) 

375 }

466fn warn_if_not_positive(py: pyo3::Python<'_>, bytes: &[u8]) -> pyo3::PyResult<()> { 

467 if bytes[0] & 0x80 != 0 || bytes == [0] { 

468 let warning_cls = types::DEPRECATED_IN_36.get(py)?; 

469 let message = c"Parsed a serial number which wasn't positive (i.e., it was negative or zero), which is disallowed by RFC 5280. Loading this certificate will cause an exception in a future release of cryptography."; 

470 pyo3::PyErr::warn(py, &warning_cls, message, 1)?; 

471 } 

472 Ok(()) 

473}

503 py: pyo3::Python<'p>, 

504 un: UserNotice<'_, Asn1Read>, 

505) -> CryptographyResult<pyo3::Bound<'p, pyo3::PyAny>> { 

506 let et = match un.explicit_text { 

507 Some(data) => parse_display_text(py, data)?, 

508 None => py.None().into_bound(py), 

509 };

511 Some(data) => { 

512 let org = parse_display_text(py, data.organization)?; 

513 let numbers = pyo3::types::PyList::empty(py); 

514 for num in data.notice_numbers.clone() { 

515 numbers.append(big_byte_slice_to_py_int(py, num.as_bytes())?)?; 

516 } 

517 types::NOTICE_REFERENCE.get(py)?.call1((org, numbers))? 

518 } 

519 None => py.None().into_bound(py), 

520 }; 

521 Ok(types::USER_NOTICE.get(py)?.call1((nr, et))?) 

522}

524fn parse_policy_qualifiers<'a>( 

525 py: pyo3::Python<'a>, 

526 policy_qualifiers: &asn1::SequenceOf<'a, PolicyQualifierInfo<'a, Asn1Read>>, 

527) -> CryptographyResult<pyo3::Bound<'a, pyo3::PyAny>> { 

528 let py_pq = pyo3::types::PyList::empty(py); 

529 for pqi in policy_qualifiers.clone() { 

530 let qualifier = match pqi.qualifier { 

531 Qualifier::CpsUri(data) => { 

532 if pqi.policy_qualifier_id == oid::CP_CPS_URI_OID { 

533 pyo3::types::PyString::new(py, data.as_str()).into_any() 

534 } else { 

535 return Err(CryptographyError::from( 

536 pyo3::exceptions::PyValueError::new_err( 

537 "CpsUri ASN.1 structure found but OID did not match", 

538 ), 

539 )); 

540 } 

541 } 

542 Qualifier::UserNotice(un) => { 

543 if pqi.policy_qualifier_id != oid::CP_USER_NOTICE_OID { 

544 return Err(CryptographyError::from( 

545 pyo3::exceptions::PyValueError::new_err( 

546 "UserNotice ASN.1 structure found but OID did not match", 

547 ), 

548 )); 

549 } 

550 parse_user_notice(py, un)? 

551 } 

552 }; 

553 py_pq.append(qualifier)?; 

554 } 

555 Ok(py_pq.into_any()) 

556}

561) -> CryptographyResult<pyo3::Bound<'p, pyo3::PyAny>> { 

562 let cp = ext.value::<asn1::SequenceOf<'_, PolicyInformation<'_, Asn1Read>>>()?; 

563 let certificate_policies = pyo3::types::PyList::empty(py); 

564 for policyinfo in cp { 

565 let pi_oid = oid_to_py_oid(py, &policyinfo.policy_identifier)?; 

566 let py_pqis = match policyinfo.policy_qualifiers { 

567 Some(policy_qualifiers) => parse_policy_qualifiers(py, &policy_qualifiers)?, 

568 None => py.None().into_bound(py), 

569 }; 

570 let pi = types::POLICY_INFORMATION 

571 .get(py)? 

572 .call1((pi_oid, py_pqis))?; 

573 certificate_policies.append(pi)?; 

574 } 

575 Ok(certificate_policies.into_any()) 

576}

635pub(crate) fn parse_distribution_point_reasons<'p>( 

636 py: pyo3::Python<'p>, 

637 reasons: Option<&asn1::BitString<'_>>, 

638) -> CryptographyResult<pyo3::Bound<'p, pyo3::PyAny>> { 

639 let reason_bit_mapping = types::REASON_BIT_MAPPING.get(py)?; 

640  

641 Ok(match reasons { 

642 Some(bs) => { 

643 let mut vec = Vec::new(); 

644 for i in 1..=8 { 

645 if bs.has_bit_set(i) { 

646 vec.push(reason_bit_mapping.get_item(i)?); 

647 } 

648 } 

649 pyo3::types::PyFrozenSet::new(py, &vec)?.into_any() 

650 } 

651 None => py.None().into_bound(py), 

652 }) 

653}

655pub(crate) fn encode_distribution_point_reasons( 

656 py: pyo3::Python<'_>, 

657 py_reasons: &pyo3::Bound<'_, pyo3::PyAny>, 

658) -> pyo3::PyResult<asn1::OwnedBitString> { 

659 let reason_flag_mapping = types::CRL_REASON_FLAGS.get(py)?; 

660  

661 let mut bits = vec![0, 0]; 

662 for py_reason in py_reasons.try_iter()? { 

663 let bit = reason_flag_mapping 

664 .get_item(py_reason?)? 

665 .extract::<usize>()?; 

666 set_bit(&mut bits, bit, true); 

667 } 

668 while bits.last() == Some(&0) { 

669 bits.pop(); 

670 } 

671 let unused_bits = bits.last().map_or(0, |b| b.trailing_zeros() as u8); 

672 Ok(asn1::OwnedBitString::new(bits, unused_bits).unwrap()) 

673}

805pub fn parse_cert_ext<'p>( 

806 py: pyo3::Python<'p>, 

807 ext: &Extension<'p>, 

808) -> CryptographyResult<Option<pyo3::Bound<'p, pyo3::PyAny>>> { 

809 match ext.extn_id { 

810 oid::SUBJECT_ALTERNATIVE_NAME_OID => { 

811 let gn_seq = ext.value::<SubjectAlternativeName<'_>>().map_err(|e| { 

812 e.add_location(asn1::ParseLocation::Field("subject_alternative_name")) 

813 })?; 

814 let sans = x509::parse_general_names(py, &gn_seq)?; 

815 Ok(Some( 

816 types::SUBJECT_ALTERNATIVE_NAME.get(py)?.call1((sans,))?, 

817 )) 

818 } 

819 oid::ISSUER_ALTERNATIVE_NAME_OID => { 

820 let gn_seq = ext.value::<IssuerAlternativeName<'_>>().map_err(|e| { 

821 e.add_location(asn1::ParseLocation::Field("issuer_alternative_name")) 

822 })?; 

823 let ians = x509::parse_general_names(py, &gn_seq)?; 

824 Ok(Some( 

825 types::ISSUER_ALTERNATIVE_NAME.get(py)?.call1((ians,))?, 

826 )) 

827 } 

828 oid::TLS_FEATURE_OID => { 

829 let tls_feature_type_to_enum = types::TLS_FEATURE_TYPE_TO_ENUM.get(py)?; 

830  

831 let features = pyo3::types::PyList::empty(py); 

832 for feature in ext.value::<asn1::SequenceOf<'_, u64>>()? { 

833 let py_feature = tls_feature_type_to_enum.get_item(feature)?; 

834 features.append(py_feature)?; 

835 } 

836 Ok(Some(types::TLS_FEATURE.get(py)?.call1((features,))?)) 

837 } 

838 oid::SUBJECT_KEY_IDENTIFIER_OID => { 

839 let identifier = ext.value::<&[u8]>()?; 

840 Ok(Some( 

841 types::SUBJECT_KEY_IDENTIFIER 

842 .get(py)? 

843 .call1((identifier,))?, 

844 )) 

845 } 

846 oid::EXTENDED_KEY_USAGE_OID => { 

847 let ekus = pyo3::types::PyList::empty(py); 

848 for oid in ext.value::<ExtendedKeyUsage<'_>>()? { 

849 let oid_obj = oid_to_py_oid(py, &oid)?; 

850 ekus.append(oid_obj)?; 

851 } 

852 Ok(Some(types::EXTENDED_KEY_USAGE.get(py)?.call1((ekus,))?)) 

853 } 

854 oid::KEY_USAGE_OID => { 

855 let kus = ext.value::<KeyUsage<'_>>()?; 

856  

857 Ok(Some(types::KEY_USAGE.get(py)?.call1(( 

858 kus.digital_signature(), 

859 kus.content_commitment(), 

860 kus.key_encipherment(), 

861 kus.data_encipherment(), 

862 kus.key_agreement(), 

863 kus.key_cert_sign(), 

864 kus.crl_sign(), 

865 kus.encipher_only(), 

866 kus.decipher_only(), 

867 ))?)) 

868 } 

869 oid::AUTHORITY_INFORMATION_ACCESS_OID => { 

870 let ads = parse_access_descriptions(py, ext)?; 

871 Ok(Some( 

872 types::AUTHORITY_INFORMATION_ACCESS.get(py)?.call1((ads,))?, 

873 )) 

874 } 

875 oid::SUBJECT_INFORMATION_ACCESS_OID => { 

876 let ads = parse_access_descriptions(py, ext)?; 

877 Ok(Some( 

878 types::SUBJECT_INFORMATION_ACCESS.get(py)?.call1((ads,))?, 

879 )) 

880 } 

881 oid::CERTIFICATE_POLICIES_OID => { 

882 let cp = parse_cp(py, ext)?; 

883 Ok(Some(types::CERTIFICATE_POLICIES.get(py)?.call1((cp,))?)) 

884 } 

885 oid::POLICY_CONSTRAINTS_OID => { 

886 let pc = ext.value::<PolicyConstraints>()?; 

887 Ok(Some(types::POLICY_CONSTRAINTS.get(py)?.call1(( 

888 pc.require_explicit_policy, 

889 pc.inhibit_policy_mapping, 

890 ))?)) 

891 } 

892 oid::OCSP_NO_CHECK_OID => { 

893 ext.value::<()>()?; 

894 Ok(Some(types::OCSP_NO_CHECK.get(py)?.call0()?)) 

895 } 

896 oid::INHIBIT_ANY_POLICY_OID => { 

897 let bignum = ext.value::<asn1::BigUint<'_>>()?; 

898 let pynum = big_byte_slice_to_py_int(py, bignum.as_bytes())?; 

899 Ok(Some(types::INHIBIT_ANY_POLICY.get(py)?.call1((pynum,))?)) 

900 } 

901 oid::BASIC_CONSTRAINTS_OID => { 

902 let bc = ext.value::<BasicConstraints>()?; 

903 Ok(Some( 

904 types::BASIC_CONSTRAINTS 

905 .get(py)? 

906 .call1((bc.ca, bc.path_length))?, 

907 )) 

908 } 

909 oid::AUTHORITY_KEY_IDENTIFIER_OID => Ok(Some(parse_authority_key_identifier(py, ext)?)), 

910 oid::CRL_DISTRIBUTION_POINTS_OID => { 

911 let dp = parse_distribution_points(py, ext)?; 

912 Ok(Some(types::CRL_DISTRIBUTION_POINTS.get(py)?.call1((dp,))?)) 

913 } 

914 oid::FRESHEST_CRL_OID => { 

915 let dp = parse_distribution_points(py, ext)?; 

916 Ok(Some(types::FRESHEST_CRL.get(py)?.call1((dp,))?)) 

917 } 

918 oid::NAME_CONSTRAINTS_OID => { 

919 let nc = ext.value::<NameConstraints<'_, Asn1Read>>()?; 

920 let permitted_subtrees = match nc.permitted_subtrees { 

921 Some(data) => parse_general_subtrees(py, data)?, 

922 None => py.None().into_bound(py), 

923 }; 

924 let excluded_subtrees = match nc.excluded_subtrees { 

925 Some(data) => parse_general_subtrees(py, data)?, 

926 None => py.None().into_bound(py), 

927 }; 

928 Ok(Some( 

929 types::NAME_CONSTRAINTS 

930 .get(py)? 

931 .call1((permitted_subtrees, excluded_subtrees))?, 

932 )) 

933 } 

934 oid::MS_CERTIFICATE_TEMPLATE => { 

935 let ms_cert_tpl = ext.value::<MSCertificateTemplate>()?; 

936 let py_oid = oid_to_py_oid(py, &ms_cert_tpl.template_id)?; 

937 Ok(Some(types::MS_CERTIFICATE_TEMPLATE.get(py)?.call1(( 

938 py_oid, 

939 ms_cert_tpl.major_version, 

940 ms_cert_tpl.minor_version, 

941 ))?)) 

942 } 

943 oid::ADMISSIONS_OID => { 

944 let admissions = ext.value::<Admissions<'_, Asn1Read>>()?; 

945 let admission_authority = match admissions.admission_authority { 

946 Some(authority) => x509::parse_general_name(py, authority)?, 

947 None => py.None().into_bound(py), 

948 }; 

949 let py_admissions = parse_admissions(py, &admissions.contents_of_admissions)?; 

950 Ok(Some( 

951 types::ADMISSIONS 

952 .get(py)? 

953 .call1((admission_authority, py_admissions))?, 

954 )) 

955 } 

956 oid::PRIVATE_KEY_USAGE_PERIOD_OID => { 

957 let pkup = ext.value::<PrivateKeyUsagePeriod>()?; 

958  

959 let not_before = match &pkup.not_before { 

960 Some(t) => { 

961 let dt = t.as_datetime(); 

962 Some(x509::datetime_to_py(py, dt)?) 

963 } 

964 None => None, 

965 }; 

966  

967 let not_after = match &pkup.not_after { 

968 Some(t) => { 

969 let dt = t.as_datetime(); 

970 Some(x509::datetime_to_py(py, dt)?) 

971 } 

972 None => None, 

973 }; 

974  

975 Ok(Some( 

976 types::PRIVATE_KEY_USAGE_PERIOD 

977 .get(py)? 

978 .call1((not_before, not_after))?, 

979 )) 

980 } 

981 _ => Ok(None), 

982 } 

983}

1003#[pyo3::pyfunction] 

1004pub(crate) fn create_x509_certificate( 

1005 py: pyo3::Python<'_>, 

1006 builder: &pyo3::Bound<'_, pyo3::PyAny>, 

1007 private_key: &pyo3::Bound<'_, pyo3::PyAny>, 

1008 hash_algorithm: &pyo3::Bound<'_, pyo3::PyAny>, 

1009 rsa_padding: &pyo3::Bound<'_, pyo3::PyAny>, 

1010 ecdsa_deterministic: Option<bool>, 

1011) -> CryptographyResult<Certificate> { 

1012 let sigalg = if private_key.is_none() { 

1013 common::AlgorithmIdentifier { 

1014 oid: asn1::DefinedByMarker::marker(), 

1015 params: common::AlgorithmParameters::Unsigned, 

1016 } 

1017 } else { 

1018 x509::sign::compute_signature_algorithm( 

1019 py, 

1020 private_key.clone(), 

1021 hash_algorithm.clone(), 

1022 rsa_padding.clone(), 

1023 )? 

1024 }; 

1025  

1026 let spki_bytes = builder 

1027 .getattr(pyo3::intern!(py, "_public_key"))? 

1028 .call_method1( 

1029 pyo3::intern!(py, "public_bytes"), 

1030 ( 

1031 crate::serialization::Encoding::DER, 

1032 crate::serialization::PublicFormat::SubjectPublicKeyInfo, 

1033 ), 

1034 )? 

1035 .extract::<pyo3::pybacked::PyBackedBytes>()?; 

1036  

1037 let py_public_key_rsa_padding = 

1038 builder.getattr(pyo3::intern!(py, "_public_key_rsa_padding"))?; 

1039 let spki = if py_public_key_rsa_padding.is_none() { 

1040 asn1::parse_single(&spki_bytes)? 

1041 } else { 

1042 // The Python layer ensures this is only ever the PSS class. 

1043 let spki = asn1::parse_single::<common::SubjectPublicKeyInfo<'_>>(&spki_bytes)?; 

1044 // id-RSASSA-PSS with the parameters absent (the unrestricted form 

1045 // from RFC 4055). 

1046 common::WithTlv::new(common::SubjectPublicKeyInfo { 

1047 algorithm: common::AlgorithmIdentifier { 

1048 oid: asn1::DefinedByMarker::marker(), 

1049 params: common::AlgorithmParameters::RsaPss(None), 

1050 }, 

1051 subject_public_key: spki.subject_public_key, 

1052 }) 

1053 }; 

1054  

1055 let py_serial = builder 

1056 .getattr(pyo3::intern!(py, "_serial_number"))? 

1057 .extract()?; 

1058  

1059 let py_issuer_name = builder.getattr(pyo3::intern!(py, "_issuer_name"))?; 

1060 let py_subject_name = builder.getattr(pyo3::intern!(py, "_subject_name"))?; 

1061 let py_not_before = builder 

1062 .getattr(pyo3::intern!(py, "_not_valid_before"))? 

1063 .extract()?; 

1064 let py_not_after = builder 

1065 .getattr(pyo3::intern!(py, "_not_valid_after"))? 

1066 .extract()?; 

1067  

1068 let ka_vec = cryptography_keepalive::KeepAlive::new(); 

1069 let ka_bytes = cryptography_keepalive::KeepAlive::new(); 

1070  

1071 let serial_bytes = py_uint_to_big_endian_bytes(py, py_serial)?; 

1072  

1073 let ka = cryptography_keepalive::KeepAlive::new(); 

1074  

1075 let tbs_cert = cryptography_x509::certificate::TbsCertificate { 

1076 version: builder 

1077 .getattr(pyo3::intern!(py, "_version"))? 

1078 .getattr(pyo3::intern!(py, "value"))? 

1079 .extract()?, 

1080 serial: asn1::BigInt::new(&serial_bytes).unwrap(), 

1081 signature_alg: sigalg.clone(), 

1082 issuer: x509::common::encode_name(py, &ka, &py_issuer_name)?, 

1083 validity: cryptography_x509::certificate::Validity { 

1084 not_before: time_from_py(py, &py_not_before)?, 

1085 not_after: time_from_py(py, &py_not_after)?, 

1086 }, 

1087 subject: x509::common::encode_name(py, &ka, &py_subject_name)?, 

1088 spki, 

1089 issuer_unique_id: None, 

1090 subject_unique_id: None, 

1091 raw_extensions: x509::common::encode_extensions( 

1092 py,
