285fn check_dsa_parameters( 

286 py: pyo3::Python<'_>, 

287 parameters: &DsaParameterNumbers, 

288) -> CryptographyResult<()> { 

289 if ![1024, 2048, 3072, 4096].contains( 

290 &parameters 

291 .p 

292 .bind(py) 

293 .call_method0("bit_length")? 

294 .extract::<usize>()?, 

295 ) { 

296 return Err(CryptographyError::from( 

297 pyo3::exceptions::PyValueError::new_err( 

298 "p must be exactly 1024, 2048, 3072, or 4096 bits long", 

299 ), 

300 )); 

301 } 

302  

303 if ![160, 224, 256].contains( 

304 &parameters 

305 .q 

306 .bind(py) 

307 .call_method0("bit_length")? 

308 .extract::<usize>()?, 

309 ) { 

310 return Err(CryptographyError::from( 

311 pyo3::exceptions::PyValueError::new_err("q must be exactly 160, 224, or 256 bits long"), 

312 )); 

313 } 

314  

315 if parameters.g.bind(py).le(1)? || parameters.g.bind(py).ge(parameters.p.bind(py))? { 

316 return Err(CryptographyError::from( 

317 pyo3::exceptions::PyValueError::new_err("g, p don't satisfy 1 < g < p."), 

318 )); 

319 } 

320  

321 Ok(()) 

322}

324fn check_dsa_private_numbers( 

325 py: pyo3::Python<'_>, 

326 numbers: &DsaPrivateNumbers, 

327) -> CryptographyResult<()> { 

328 let params = numbers.public_numbers.get().parameter_numbers.get(); 

329 check_dsa_parameters(py, params)?; 

330  

331 if numbers.x.bind(py).le(0)? || numbers.x.bind(py).ge(params.q.bind(py))? { 

332 return Err(CryptographyError::from( 

333 pyo3::exceptions::PyValueError::new_err("x must be > 0 and < q."), 

334 )); 

335 } 

336  

337 if (**numbers.public_numbers.get().y.bind(py)).ne(params 

338 .g 

339 .bind(py) 

340 .pow(numbers.x.bind(py), Some(params.p.bind(py)))?)? 

341 { 

342 return Err(CryptographyError::from( 

343 pyo3::exceptions::PyValueError::new_err("y must be equal to (g ** x % p)."), 

344 )); 

345 } 

346  

347 Ok(()) 

348}
