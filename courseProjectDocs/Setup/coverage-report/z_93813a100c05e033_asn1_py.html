34@dataclasses.dataclass(frozen=True) 

35class Variant(typing.Generic[U, Tag]): 

36 """ 

37 A tagged variant for CHOICE fields with the same underlying type. 

38 

39 Use this when you have multiple CHOICE alternatives with the same type 

40 and need to distinguish between them: 

41 

42 foo: ( 

43 Annotated[Variant[int, typing.Literal["IntA"]], Implicit(0)] 

44 | Annotated[Variant[int, typing.Literal["IntB"]], Implicit(1)] 

45 ) 

46 

47 Usage: 

48 example = Example(foo=Variant(5, "IntA")) 

49 decoded.foo.value # The int value 

50 decoded.foo.tag # "IntA" or "IntB" 

51 """

54 tag: str

89 union_types = (

97def _resolve_type_aliases(field_type: typing.Any) -> typing.Any: 

98 # Recursively resolve PEP 695 (`type X = ...`) type aliases (Python 

99 # 3.12+) to their underlying value, so that the rest of the 

100 # normalization logic never encounters an alias. Aliases can refer 

101 # to other aliases and can appear at any level of nesting (e.g. 

102 # inside `Annotated[...]`, unions, or `list[...]`). 

103 if sys.version_info < (3, 12): 

104 return field_type 

105 

106 while isinstance(field_type, typing.TypeAliasType): 

107 field_type = field_type.__value__ 

108 

109 args = typing.get_args(field_type) 

110 resolved_args = tuple(_resolve_type_aliases(arg) for arg in args) 

111 if resolved_args == args: 

112 # No aliases anywhere inside: return the type unchanged. 

113 return field_type 

114 

115 if _is_union(field_type): 

116 # `X | Y` unions can't be rebuilt through their origin like 

117 # other generics below: `typing.get_origin` returns 

118 # `types.UnionType` for them, which is only subscriptable on 

119 # Python 3.14+. Rebuilding through `typing.Union` also 

120 # flattens any nested union introduced by an alias of a union 

121 # (e.g. `Time | int` where `type Time = UTCTime | 

122 # GeneralizedTime`), just like `typing.Union` would have done 

123 # if the alias had been written inline. 

124 return typing.Union[resolved_args] 

125 

126 # An alias appeared inside a generic (e.g. `Annotated[Time, ...]`, 

127 # `list[MyInt]`, or `SetOf[MyInt]`): re-parameterize the generic 

128 # with the resolved arguments. Subscripting with a tuple is 

129 # equivalent to subscripting with multiple arguments. 

130 return typing.get_origin(field_type)[resolved_args]

133def _extract_annotation( 

134 metadata: tuple, field_name: str 

135) -> declarative_asn1.Annotation: 

136 default = None 

137 encoding = None 

138 size = None 

139 for raw_annotation in metadata: 

140 if isinstance(raw_annotation, Default): 

141 if default is not None: 

142 raise TypeError( 

143 f"multiple DEFAULT annotations found in field " 

144 f"'{field_name}'" 

145 ) 

146 default = raw_annotation.value 

147 elif isinstance(raw_annotation, declarative_asn1.Encoding): 

148 if encoding is not None: 

149 raise TypeError( 

150 f"multiple IMPLICIT/EXPLICIT annotations found in field " 

151 f"'{field_name}'" 

152 ) 

153 encoding = raw_annotation 

154 elif isinstance(raw_annotation, declarative_asn1.Size): 

155 if size is not None: 

156 raise TypeError( 

157 f"multiple SIZE annotations found in field '{field_name}'" 

158 ) 

159 size = raw_annotation 

160 else: 

161 raise TypeError(f"unsupported annotation: {raw_annotation}") 

162 

163 return declarative_asn1.Annotation( 

164 default=default, encoding=encoding, size=size 

165 )

168def _normalize_field_type( 

169 field_type: typing.Any, field_name: str 

170) -> declarative_asn1.AnnotatedType: 

171 field_type = _resolve_type_aliases(field_type) 

172 

173 # Strip the `Annotated[...]` off, and populate the annotation 

174 # from it if it exists. 

175 if typing.get_origin(field_type) is typing.Annotated: 

176 annotation = _extract_annotation(field_type.__metadata__, field_name) 

177 field_type, *_ = typing.get_args(field_type) 

178 else: 

179 annotation = declarative_asn1.Annotation() 

180 

181 if annotation.size is not None and ( 

182 typing.get_origin(field_type) not in (builtins.list, SetOf) 

183 and field_type 

184 not in ( 

185 builtins.bytes, 

186 builtins.str, 

187 BitString, 

188 IA5String, 

189 PrintableString, 

190 ) 

191 ): 

192 raise TypeError( 

193 f"field '{field_name}' has a SIZE annotation, but SIZE " 

194 "annotations are only supported for fields of types: " 

195 "[SEQUENCE OF, SET OF, BIT STRING, OCTET STRING, UTF8String, " 

196 "PrintableString, IA5String]" 

197 ) 

198 

199 if field_type is TLV: 

200 if isinstance(annotation.encoding, Implicit): 

201 raise TypeError( 

202 f"field '{field_name}' has an IMPLICIT annotation, but " 

203 "IMPLICIT annotations are not supported for TLV types." 

204 ) 

205 elif annotation.default is not None: 

206 raise TypeError( 

207 f"field '{field_name}' has a DEFAULT annotation, but " 

208 "DEFAULT annotations are not supported for TLV types." 

209 ) 

210 

211 _check_x509_field_annotations(field_type, annotation, field_name) 

212 

213 if hasattr(field_type, "__asn1_root__"): 

214 root_type = field_type.__asn1_root__ 

215 if not isinstance( 

216 root_type, 

217 ( 

218 declarative_asn1.Type.Sequence, 

219 declarative_asn1.Type.Set, 

220 declarative_asn1.Type.ValueSet, 

221 ), 

222 ): 

223 raise TypeError(f"unsupported root type: {root_type}") 

224 return declarative_asn1.AnnotatedType( 

225 typing.cast(declarative_asn1.Type, root_type), annotation 

226 ) 

227 elif _is_union(field_type): 

228 union_args = typing.get_args(field_type) 

229 if len(union_args) == 2 and NoneType in union_args: 

230 # A Union between a type and None is an OPTIONAL 

231 optional_type = ( 

232 union_args[0] if union_args[1] is type(None) else union_args[1] 

233 ) 

234 if optional_type is TLV: 

235 raise TypeError( 

236 "optional TLV types (`TLV | None`) are not " 

237 "currently supported" 

238 ) 

239 # For optional types, the annotation is associated with the 

240 # union, so we check it against the inner type here. 

241 _check_x509_field_annotations( 

242 optional_type, annotation, field_name 

243 ) 

244 annotated_type = _normalize_field_type(optional_type, field_name) 

245 

246 if not annotated_type.annotation.is_empty(): 

247 raise TypeError( 

248 "optional (`X | None`) types cannot have `X` " 

249 "annotated: annotations must apply to the union " 

250 "(i.e: `Annotated[X | None, annotation]`)" 

251 ) 

252 

253 if annotation.default is not None: 

254 raise TypeError( 

255 "optional (`X | None`) types should not have a DEFAULT " 

256 "annotation" 

257 ) 

258 

259 rust_field_type = declarative_asn1.Type.Option(annotated_type) 

260 

261 else: 

262 # Otherwise, the Union is a CHOICE 

263 if isinstance(annotation.encoding, Implicit): 

264 # CHOICEs cannot be IMPLICIT. See X.680 section 31.2.9. 

265 raise TypeError( 

266 "CHOICE (`X | Y | ...`) types should not have an IMPLICIT " 

267 "annotation" 

268 ) 

269 variants = [ 

270 _type_to_variant(arg, field_name) 

271 for arg in union_args 

272 if arg is not type(None) 

273 ] 

274 

275 # Union types should either be all Variants 

276 # (`Variant[..] | Variant[..] | etc`) or all non Variants 

277 are_union_types_tagged = variants[0].tag_name is not None 

278 if any( 

279 (v.tag_name is not None) != are_union_types_tagged 

280 for v in variants 

281 ): 

282 raise TypeError( 

283 "When using `asn1.Variant` in a union, all the other " 

284 "types in the union must also be `asn1.Variant`" 

285 ) 

286 

287 if are_union_types_tagged: 

288 tags = {v.tag_name for v in variants} 

289 if len(variants) != len(tags): 

290 raise TypeError( 

291 "When using `asn1.Variant` in a union, the tags used " 

292 "must be unique" 

293 ) 

294 

295 rust_choice_type = declarative_asn1.Type.Choice(variants) 

296 # If None is part of the union types, this is an OPTIONAL CHOICE 

297 rust_field_type = ( 

298 declarative_asn1.Type.Option( 

299 declarative_asn1.AnnotatedType( 

300 rust_choice_type, declarative_asn1.Annotation() 

301 ) 

302 ) 

303 if NoneType in union_args 

304 else rust_choice_type 

305 ) 

306 

307 elif typing.get_origin(field_type) is builtins.list: 

308 inner_type = _normalize_field_type( 

309 typing.get_args(field_type)[0], field_name 

310 ) 

311 _check_collection_element_type(inner_type, field_name, "list") 

312 rust_field_type = declarative_asn1.Type.SequenceOf(inner_type) 

313 elif typing.get_origin(field_type) is SetOf: 

314 inner_type = _normalize_field_type( 

315 typing.get_args(field_type)[0], field_name 

316 ) 

317 _check_collection_element_type(inner_type, field_name, "SetOf") 

318 rust_field_type = declarative_asn1.Type.SetOf(inner_type) 

319 else: 

320 rust_field_type = declarative_asn1.non_root_python_to_rust(field_type) 

321 

322 return declarative_asn1.AnnotatedType(rust_field_type, annotation)

326# a DEFAULT: `list[X | None]` and `SetOf[X | None]` are not valid (the 

327# collection as a whole can be, as `list[X] | None`). Such an element would 

328# also decode to a value without consuming any input, so decoding could never 

329# make progress.

331 element_type: declarative_asn1.AnnotatedType, 

332 field_name: str, 

333 collection_name: str, 

334) -> None:

336 isinstance(element_type.inner, declarative_asn1.Type.Option) 

337 or element_type.annotation.default is not None 

338 ): 

339 raise TypeError( 

340 f"field '{field_name}': the element type of a `{collection_name}`"

345# Convert a type to a Variant. Used with types inside Union 

346# annotations (T1, T2, etc in `Union[T1, T2, ...]`). 

347def _type_to_variant( 

348 t: typing.Any, field_name: str 

349) -> declarative_asn1.Variant: 

350 is_annotated = typing.get_origin(t) is typing.Annotated 

351 inner_type = typing.get_args(t)[0] if is_annotated else t 

352 

353 # Check if this is a Variant[T, Tag] type 

354 if typing.get_origin(inner_type) is Variant: 

355 value_type, tag_literal = typing.get_args(inner_type) 

356 if typing.get_origin(tag_literal) is not typing.Literal: 

357 raise TypeError( 

358 "When using `asn1.Variant` in a type annotation, the second " 

359 "type parameter must be a `typing.Literal` type. E.g: " 

360 '`Variant[int, typing.Literal["MyInt"]]`.' 

361 ) 

362 tag_name = typing.get_args(tag_literal)[0] 

363 

364 if hasattr(value_type, "__asn1_root__"): 

365 rust_type = value_type.__asn1_root__ 

366 else: 

367 rust_type = declarative_asn1.non_root_python_to_rust(value_type) 

368 

369 if is_annotated: 

370 ann_type = declarative_asn1.AnnotatedType( 

371 rust_type, 

372 _extract_annotation(t.__metadata__, field_name), 

373 ) 

374 else: 

375 ann_type = declarative_asn1.AnnotatedType( 

376 rust_type, 

377 declarative_asn1.Annotation(), 

378 ) 

379 

380 return declarative_asn1.Variant(Variant, ann_type, tag_name) 

381 else: 

382 # Plain type (not a tagged Variant) 

383 return declarative_asn1.Variant( 

384 inner_type, 

385 _normalize_field_type(t, field_name), 

386 None, 

387 )

390def _annotate_fields( 

391 raw_fields: dict[str, type], 

392) -> dict[str, declarative_asn1.AnnotatedType]: 

393 fields = {}

395 # Recursively normalize the field type into something that the 

396 # Rust code can understand. 

397 annotated_field_type = _normalize_field_type(field_type, field_name) 

398 fields[field_name] = annotated_field_type 

399 

400 return fields

498def value_set( 

499 value_type: type, 

500) -> typing.Callable[[type[U]], type[U]]: 

501 """ 

502 A class decorator that registers an `enum.Enum` subclass as an 

503 ASN.1 value set of the given underlying type. All the member 

504 values must be instances of `value_type`. Members are encoded as 

505 their value; decoding fails if the decoded value does not match 

506 any member. 

507 """ 

508 rust_type = declarative_asn1.non_root_python_to_rust(value_type) 

509 

510 def decorator(cls: type[U]) -> type[U]: 

511 if not issubclass(cls, enum.Enum): 

512 raise TypeError( 

513 "value sets can only be defined from enum.Enum subclasses" 

514 ) 

515 members = list(cls) 

516 if not members: 

517 raise TypeError( 

518 f"value set '{cls.__name__}' must have at least one member" 

519 ) 

520 for member in members: 

521 if not isinstance(member.value, value_type): 

522 raise TypeError( 

523 f"member '{member.name}' of value set '{cls.__name__}' " 

524 f"must have a value of type " 

525 f"'{value_type.__name__}', got: " 

526 f"'{type(member.value).__name__}'" 

527 ) 

528 inner = declarative_asn1.AnnotatedType( 

529 rust_type, declarative_asn1.Annotation() 

530 ) 

531 # Map from member value to member, used for O(1) lookups when 

532 # decoding. This requires the member values to be hashable. 

533 value_map = {member.value: member for member in members} 

534 root = declarative_asn1.Type.ValueSet(cls, inner, value_map) 

535 

536 setattr(cls, "__asn1_root__", root) 

537 return cls 

538 

539 return decorator
