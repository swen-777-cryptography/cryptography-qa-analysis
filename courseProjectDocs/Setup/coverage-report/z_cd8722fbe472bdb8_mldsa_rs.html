47/// Check whether a PKey is an ML-DSA key. 

48/// 

49/// OpenSSL 3.x provider-based keys return -1 from EVP_PKEY_get_id(), so 

50/// NID-based matching does not work. `PKeyRef::is_a` queries by algorithm 

51/// name instead. 

52#[cfg(any( 

53 CRYPTOGRAPHY_IS_BORINGSSL, 

54 CRYPTOGRAPHY_IS_AWSLC, 

55 CRYPTOGRAPHY_OPENSSL_350_OR_GREATER 

56))] 

57pub fn is_mldsa_pkey<T>(pkey: &openssl::pkey::PKeyRef<T>) -> bool { 

58 cfg_if::cfg_if! { 

59 if #[cfg(CRYPTOGRAPHY_IS_BORINGSSL)] { 

60 let raw = pkey.id().as_raw(); 

61 raw == ffi::NID_ML_DSA_44 || raw == ffi::NID_ML_DSA_65 || raw == ffi::NID_ML_DSA_87 

62 } else if #[cfg(CRYPTOGRAPHY_IS_AWSLC)] { 

63 pkey.id() == PKEY_ID 

64 } else if #[cfg(CRYPTOGRAPHY_OPENSSL_350_OR_GREATER)] { 

65 pkey.is_a(openssl::pkey::KeyType::ML_DSA_44) 

66 || pkey.is_a(openssl::pkey::KeyType::ML_DSA_65) 

67 || pkey.is_a(openssl::pkey::KeyType::ML_DSA_87) 

68 } 

69 } 

70}

82 pub fn from_pkey<T: openssl::pkey::HasPublic>( 

83 pkey: &openssl::pkey::PKeyRef<T>, 

84 ) -> MlDsaVariant { 

85 cfg_if::cfg_if! { 

86 if #[cfg(CRYPTOGRAPHY_IS_BORINGSSL)] { 

87 match pkey.id().as_raw() { 

88 ffi::NID_ML_DSA_44 => MlDsaVariant::MlDsa44, 

89 ffi::NID_ML_DSA_65 => MlDsaVariant::MlDsa65, 

90 ffi::NID_ML_DSA_87 => MlDsaVariant::MlDsa87, 

91 _ => panic!("Unsupported ML-DSA variant"), 

92 } 

93 } else if #[cfg(CRYPTOGRAPHY_IS_AWSLC)] { 

94 // SAFETY: EVP_PKEY_pqdsa_get_type returns the NID of the 

95 // PQDSA algorithm for a valid PQDSA pkey. 

96 let nid = unsafe { ffi::EVP_PKEY_pqdsa_get_type(pkey.as_ptr()) }; 

97 match nid { 

98 ffi::NID_MLDSA44 => MlDsaVariant::MlDsa44, 

99 ffi::NID_MLDSA65 => MlDsaVariant::MlDsa65, 

100 ffi::NID_MLDSA87 => MlDsaVariant::MlDsa87, 

101 _ => panic!("Unsupported ML-DSA variant"), 

102 } 

103 } else if #[cfg(CRYPTOGRAPHY_OPENSSL_350_OR_GREATER)] { 

104 // Provider-based keys in OpenSSL 3.x return -1 from 

105 // EVP_PKEY_get_id(), so we must use name-based lookup. 

106 if pkey.is_a(openssl::pkey::KeyType::ML_DSA_44) { 

107 MlDsaVariant::MlDsa44 

108 } else if pkey.is_a(openssl::pkey::KeyType::ML_DSA_65) { 

109 MlDsaVariant::MlDsa65 

110 } else if pkey.is_a(openssl::pkey::KeyType::ML_DSA_87) { 

111 MlDsaVariant::MlDsa87 

112 } else { 

113 panic!("Unsupported ML-DSA variant") 

114 } 

115 } 

116 } 

117 }

155/// Extract the raw 32-byte seed from an ML-DSA private key. 

156pub fn mldsa_seed_raw( 

157 pkey: &openssl::pkey::PKeyRef<openssl::pkey::Private>, 

158) -> OpenSSLResult<[u8; 32]> { 

159 let mut seed = [0u8; 32]; 

160 cfg_if::cfg_if! { 

161 if #[cfg(any(CRYPTOGRAPHY_IS_BORINGSSL, CRYPTOGRAPHY_IS_AWSLC))] { 

162 let mut seed_len = seed.len(); 

163 // SAFETY: pkey is a valid EVP_PKEY and seed is a 32-byte buffer. 

164 unsafe { 

165 cvt(ffi::EVP_PKEY_get_private_seed( 

166 pkey.as_ptr(), 

167 seed.as_mut_ptr(), 

168 &mut seed_len, 

169 ))?; 

170 } 

171 assert_eq!(seed_len, 32); 

172 } else if #[cfg(CRYPTOGRAPHY_OPENSSL_350_OR_GREATER)] { 

173 pkey.seed_into(&mut seed)?; 

174 } 

175 } 

176 Ok(seed) 

177}

255pub fn sign( 

256 pkey: &openssl::pkey::PKeyRef<openssl::pkey::Private>, 

257 data: &[u8], 

258 context: &[u8], 

259) -> OpenSSLResult<Vec<u8>> { 

260 let mut md_ctx = openssl::md_ctx::MdCtx::new()?; 

261 let pkey_ctx = md_ctx.digest_sign_init(None, pkey)?; 

262 if !context.is_empty() { 

263 set_context_string(pkey_ctx, context)?; 

264 } 

265 let mut sig = vec![]; 

266 md_ctx.digest_sign_to_vec(data, &mut sig)?; 

267 Ok(sig) 

268}

284/// Enable "external mu" mode on an OpenSSL signing/verification context by 

285/// setting the integer `mu` signature parameter (OSSL_SIGNATURE_PARAM_MU) to 1. 

286#[cfg(CRYPTOGRAPHY_OPENSSL_350_OR_GREATER)] 

287fn set_mu<T>(pkey_ctx: &mut openssl::pkey_ctx::PkeyCtxRef<T>) -> OpenSSLResult<()> { 

288 // A fixed OSSL_PARAM array holding the integer "mu" parameter set to 1 

289 // enables external mu mode. The provider reads it back with 

290 // OSSL_PARAM_get_int, which accepts an unsigned integer param, so we can 

291 // build the array on the stack rather than via an allocating 

292 // OSSL_PARAM_BLD. 

293 let mut mu: std::ffi::c_uint = 1; 

294 // SAFETY: `params` and its backing `mu` value outlive the call into 

295 // OpenSSL, and the array is terminated with OSSL_PARAM_construct_end(). 

296 unsafe { 

297 let params = [ 

298 ffi::OSSL_PARAM_construct_uint(c"mu".as_ptr(), &mut mu), 

299 ffi::OSSL_PARAM_construct_end(), 

300 ]; 

301 cvt(ffi::EVP_PKEY_CTX_set_params( 

302 pkey_ctx.as_ptr(), 

303 params.as_ptr(), 

304 ))?; 

305 } 

306 Ok(()) 

307}

309/// Sign a precomputed external mu (message representative). `mu` must be 

310/// [`MLDSA_MU_BYTES`] long, and already incorporates any context string, so no 

311/// context is accepted here. 

312pub fn sign_mu( 

313 pkey: &openssl::pkey::PKeyRef<openssl::pkey::Private>, 

314 mu: &[u8], 

315) -> OpenSSLResult<Vec<u8>> { 

316 // The BoringSSL path passes `mu` to the low-level API as a bare pointer 

317 // that is read for exactly MLDSA_MU_BYTES, so enforce the length here for 

318 // soundness rather than relying on callers. 

319 assert_eq!(mu.len(), MLDSA_MU_BYTES); 

320 cfg_if::cfg_if! { 

321 if #[cfg(CRYPTOGRAPHY_IS_BORINGSSL)] { 

322 // BoringSSL has no EVP-level external mu support, so we drop down to 

323 // the low-level ML-DSA API, reconstructing the private key from its 

324 // 32-byte seed. 

325 let seed = mldsa_seed_raw(pkey)?; 

326 // SAFETY: `seed` is a valid 32-byte seed and `mu` is a valid 

327 // MLDSA_MU_BYTES buffer; both outlive the calls below. 

328 unsafe { 

329 match MlDsaVariant::from_pkey(pkey) { 

330 MlDsaVariant::MlDsa44 => { 

331 let mut key = std::mem::MaybeUninit::<ffi::MLDSA44_private_key>::uninit(); 

332 cvt(ffi::MLDSA44_private_key_from_seed( 

333 key.as_mut_ptr(), 

334 seed.as_ptr(), 

335 seed.len(), 

336 ))?; 

337 let key = key.assume_init(); 

338 let mut sig = vec![0u8; ffi::MLDSA44_SIGNATURE_BYTES as usize]; 

339 cvt(ffi::MLDSA44_sign_message_representative( 

340 sig.as_mut_ptr(), 

341 &key, 

342 mu.as_ptr(), 

343 ))?; 

344 Ok(sig) 

345 } 

346 MlDsaVariant::MlDsa65 => { 

347 let mut key = std::mem::MaybeUninit::<ffi::MLDSA65_private_key>::uninit(); 

348 cvt(ffi::MLDSA65_private_key_from_seed( 

349 key.as_mut_ptr(), 

350 seed.as_ptr(), 

351 seed.len(), 

352 ))?; 

353 let key = key.assume_init(); 

354 let mut sig = vec![0u8; ffi::MLDSA65_SIGNATURE_BYTES as usize]; 

355 cvt(ffi::MLDSA65_sign_message_representative( 

356 sig.as_mut_ptr(), 

357 &key, 

358 mu.as_ptr(), 

359 ))?; 

360 Ok(sig) 

361 } 

362 MlDsaVariant::MlDsa87 => { 

363 let mut key = std::mem::MaybeUninit::<ffi::MLDSA87_private_key>::uninit(); 

364 cvt(ffi::MLDSA87_private_key_from_seed( 

365 key.as_mut_ptr(), 

366 seed.as_ptr(), 

367 seed.len(), 

368 ))?; 

369 let key = key.assume_init(); 

370 let mut sig = vec![0u8; ffi::MLDSA87_SIGNATURE_BYTES as usize]; 

371 cvt(ffi::MLDSA87_sign_message_representative( 

372 sig.as_mut_ptr(), 

373 &key, 

374 mu.as_ptr(), 

375 ))?; 

376 Ok(sig) 

377 } 

378 } 

379 } 

380 } else if #[cfg(CRYPTOGRAPHY_IS_AWSLC)] { 

381 // AWS-LC's EVP_PKEY_sign treats its input as an external mu (the 

382 // "ExternalMu" format) for ML-DSA keys. 

383 let mut ctx = openssl::pkey_ctx::PkeyCtx::new(pkey)?; 

384 ctx.sign_init()?; 

385 let mut sig = vec![]; 

386 ctx.sign_to_vec(mu, &mut sig)?; 

387 Ok(sig) 

388 } else if #[cfg(CRYPTOGRAPHY_OPENSSL_350_OR_GREATER)] { 

389 // OpenSSL signs an external mu by setting the "mu" parameter and 

390 // passing the 64-byte mu in place of the message. ML-DSA only 

391 // supports the digest-sign flow (EVP_PKEY_sign_init is not 

392 // implemented for it), so we cannot use the one-shot PkeyCtx path 

393 // the AWS-LC branch uses. 

394 let mut md_ctx = openssl::md_ctx::MdCtx::new()?; 

395 let pkey_ctx = md_ctx.digest_sign_init(None, pkey)?; 

396 set_mu(pkey_ctx)?; 

397 let mut sig = vec![]; 

398 md_ctx.digest_sign_to_vec(mu, &mut sig)?; 

399 Ok(sig) 

400 } 

401 } 

402}

404/// Verify a signature over a precomputed external mu (message representative). 

405/// `mu` must be [`MLDSA_MU_BYTES`] long. 

406pub fn verify_mu( 

407 pkey: &openssl::pkey::PKeyRef<openssl::pkey::Public>, 

408 signature: &[u8], 

409 mu: &[u8], 

410) -> OpenSSLResult<bool> { 

411 // The BoringSSL path passes `mu` to the low-level API as a bare pointer 

412 // that is read for exactly MLDSA_MU_BYTES, so enforce the length here for 

413 // soundness rather than relying on callers. 

414 assert_eq!(mu.len(), MLDSA_MU_BYTES); 

415 cfg_if::cfg_if! { 

416 if #[cfg(CRYPTOGRAPHY_IS_BORINGSSL)] { 

417 let raw = pkey.raw_public_key()?; 

418 // SAFETY: We parse the low-level public key from its encoded form 

419 // and verify the signature over the MLDSA_MU_BYTES `mu`. 

420 unsafe { 

421 match MlDsaVariant::from_pkey(pkey) { 

422 MlDsaVariant::MlDsa44 => { 

423 let mut key = std::mem::MaybeUninit::<ffi::MLDSA44_public_key>::uninit(); 

424 let mut cbs = std::mem::MaybeUninit::<ffi::CBS>::uninit(); 

425 ffi::CBS_init(cbs.as_mut_ptr(), raw.as_ptr(), raw.len()); 

426 cvt(ffi::MLDSA44_parse_public_key( 

427 key.as_mut_ptr(), 

428 cbs.as_mut_ptr(), 

429 ))?; 

430 let key = key.assume_init(); 

431 Ok(ffi::MLDSA44_verify_message_representative( 

432 &key, 

433 signature.as_ptr(), 

434 signature.len(), 

435 mu.as_ptr(), 

436 ) == 1) 

437 } 

438 MlDsaVariant::MlDsa65 => { 

439 let mut key = std::mem::MaybeUninit::<ffi::MLDSA65_public_key>::uninit(); 

440 let mut cbs = std::mem::MaybeUninit::<ffi::CBS>::uninit(); 

441 ffi::CBS_init(cbs.as_mut_ptr(), raw.as_ptr(), raw.len()); 

442 cvt(ffi::MLDSA65_parse_public_key( 

443 key.as_mut_ptr(), 

444 cbs.as_mut_ptr(), 

445 ))?; 

446 let key = key.assume_init(); 

447 Ok(ffi::MLDSA65_verify_message_representative( 

448 &key, 

449 signature.as_ptr(), 

450 signature.len(), 

451 mu.as_ptr(), 

452 ) == 1) 

453 } 

454 MlDsaVariant::MlDsa87 => { 

455 let mut key = std::mem::MaybeUninit::<ffi::MLDSA87_public_key>::uninit(); 

456 let mut cbs = std::mem::MaybeUninit::<ffi::CBS>::uninit(); 

457 ffi::CBS_init(cbs.as_mut_ptr(), raw.as_ptr(), raw.len()); 

458 cvt(ffi::MLDSA87_parse_public_key( 

459 key.as_mut_ptr(), 

460 cbs.as_mut_ptr(), 

461 ))?; 

462 let key = key.assume_init(); 

463 Ok(ffi::MLDSA87_verify_message_representative( 

464 &key, 

465 signature.as_ptr(), 

466 signature.len(), 

467 mu.as_ptr(), 

468 ) == 1) 

469 } 

470 } 

471 } 

472 } else if #[cfg(CRYPTOGRAPHY_IS_AWSLC)] { 

473 let mut ctx = openssl::pkey_ctx::PkeyCtx::new(pkey)?; 

474 ctx.verify_init()?; 

475 Ok(ctx.verify(mu, signature).unwrap_or(false)) 

476 } else if #[cfg(CRYPTOGRAPHY_OPENSSL_350_OR_GREATER)] { 

477 let mut md_ctx = openssl::md_ctx::MdCtx::new()?; 

478 let pkey_ctx = md_ctx.digest_verify_init(None, pkey)?; 

479 set_mu(pkey_ctx)?; 

480 Ok(md_ctx.digest_verify(mu, signature).unwrap_or(false)) 

481 } 

482 }
