528impl SetOf { 

529 #[new]

532 SetOf { inner }

539 fn __eq__(&self, py: pyo3::Python<'_>, other: pyo3::PyRef<'_, Self>) -> pyo3::PyResult<bool> { 

540 // SET OF is unordered, so two values are equal if they contain the 

541 // same elements in any order. Elements needn't be hashable or 

542 // orderable, so pair them up with `==` (this is quadratic, but 

543 // these are small). 

544 let this = self.inner.bind(py); 

545 let other = other.inner.bind(py); 

546 if this.len() != other.len() { 

547 return Ok(false); 

548 } 

549 let mut unmatched: Vec<pyo3::Bound<'_, pyo3::PyAny>> = other.iter().collect(); 

550 for item in this.iter() { 

551 let mut found = None; 

552 for (i, candidate) in unmatched.iter().enumerate() { 

553 if item.eq(candidate)? { 

554 found = Some(i); 

555 break; 

556 } 

557 } 

558 match found { 

559 Some(i) => { 

560 unmatched.swap_remove(i); 

561 } 

562 None => return Ok(false), 

563 } 

564 } 

565 Ok(true) 

566 }

569 Ok(format!("SetOf({})", self.inner.bind(py).repr()?)) 

570 } 

571}

573/// Utility function for converting builtin Python types

575#[pyo3::pyfunction] 

576pub fn non_root_python_to_rust<'p>( 

577 py: pyo3::Python<'p>, 

578 class: &pyo3::Bound<'p, pyo3::types::PyType>, 

579) -> pyo3::PyResult<pyo3::Bound<'p, Type>> { 

580 if class.is(pyo3::types::PyInt::type_object(py)) { 

581 Type::PyInt().into_pyobject(py) 

582 } else if class.is(pyo3::types::PyBool::type_object(py)) { 

583 Type::PyBool().into_pyobject(py) 

584 } else if class.is(pyo3::types::PyString::type_object(py)) { 

585 Type::PyStr().into_pyobject(py) 

586 } else if class.is(pyo3::types::PyBytes::type_object(py)) { 

587 Type::PyBytes().into_pyobject(py) 

588 } else if class.is(PrintableString::type_object(py)) { 

589 Type::PrintableString().into_pyobject(py) 

590 } else if class.is(IA5String::type_object(py)) { 

591 Type::IA5String().into_pyobject(py) 

592 } else if class.is(crate::oid::ObjectIdentifier::type_object(py)) { 

593 Type::ObjectIdentifier().into_pyobject(py) 

594 } else if class.is(UtcTime::type_object(py)) { 

595 Type::UtcTime().into_pyobject(py) 

596 } else if class.is(GeneralizedTime::type_object(py)) { 

597 Type::GeneralizedTime().into_pyobject(py) 

598 } else if class.is(BitString::type_object(py)) { 

599 Type::BitString().into_pyobject(py) 

600 } else if class.is(Tlv::type_object(py)) { 

601 Type::Tlv().into_pyobject(py) 

602 } else if class.is(Null::type_object(py)) { 

603 Type::Null().into_pyobject(py) 

604 } else if class.is(crate::x509::certificate::Certificate::type_object(py)) { 

605 Type::Certificate().into_pyobject(py) 

606 } else if class.is(crate::x509::csr::CertificateSigningRequest::type_object(py)) {

608 } else if class.is(crate::x509::crl::CertificateRevocationList::type_object(py)) {

610 } else if class.is(&crate::types::NAME.get(py)?) { 

611 Type::Name().into_pyobject(py) 

612 } else { 

613 Err(pyo3::exceptions::PyTypeError::new_err(format!( 

614 "cannot handle type: {class:?}" 

615 ))) 

616 }

633 encoding: None,

635 } 

636 .into_pyobject(py)? 

637 .unbind(), 

638 })

641// Utility function for converting a Python class or a Python builtin type 

642// into an AnnotatedType. 

643pub(crate) fn python_class_to_annotated<'p>( 

644 py: pyo3::Python<'p>, 

645 class: &pyo3::Bound<'p, pyo3::types::PyType>, 

646) -> pyo3::PyResult<pyo3::Bound<'p, AnnotatedType>> { 

647 if let Ok(root) = class.getattr("__asn1_root__") { 

648 // Handle decorated classes 

649 AnnotatedType { 

650 inner: root.cast_into::<Type>()?.unbind(), 

651 annotation: Annotation { 

652 default: None, 

653 encoding: None, 

654 size: None, 

655 } 

656 .into_pyobject(py)? 

657 .unbind(), 

658 } 

659 .into_pyobject(py) 

660 } else { 

661 // Handle builtin types 

662 pyo3::Bound::new(py, non_root_type_to_annotated(py, class)?) 

663 } 

664}

666// Builds the `AnnotatedType` used to encode a top-level value passed to 

667// `encode_der`. 

668pub(crate) fn encode_value_to_annotated<'p>( 

669 py: pyo3::Python<'p>, 

670 value: &pyo3::Bound<'p, pyo3::types::PyAny>, 

671) -> pyo3::PyResult<pyo3::Bound<'p, AnnotatedType>> { 

672 if let Ok(setof) = value.cast::<SetOf>() { 

673 // SET OF is homogeneous, so we infer the element type from the 

674 // first element. 

675 let inner = match setof.get().inner.bind(py).iter().next() { 

676 Some(first) => { 

677 let class = first.get_type(); 

678 python_class_to_annotated(py, &class)? 

679 } 

680 // An empty SET OF has no elements to encode, so the inner 

681 // type is never used; any type works as a placeholder. 

682 None => python_class_to_annotated(py, &Tlv::type_object(py))?, 

683 }; 

684 return pyo3::Bound::new( 

685 py, 

686 AnnotatedType { 

687 inner: pyo3::Py::new(py, Type::SetOf(inner.unbind()))?, 

688 annotation: Annotation { 

689 default: None, 

690 encoding: None, 

691 size: None, 

692 } 

693 .into_pyobject(py)? 

694 .unbind(), 

695 }, 

696 ); 

697 } 

698 python_class_to_annotated(py, &value.get_type()) 

699}

701// Checks if encoding `tag_without_encoding` using `encoding` results 

702// in `tag` 

703fn check_tag_with_encoding( 

704 tag_without_encoding: asn1::Tag, 

705 encoding: &Option<pyo3::Py<Encoding>>, 

706 tag: asn1::Tag, 

707) -> bool { 

708 let tag_with_encoding = match encoding { 

709 Some(e) => match e.get() { 

710 Encoding::Implicit(n) => asn1::implicit_tag(*n, tag_without_encoding), 

711 Encoding::Explicit(n) => asn1::explicit_tag(*n), 

712 }, 

713 None => tag_without_encoding, 

714 }; 

715 tag_with_encoding == tag 

716}

718// Utility function to see if a tag matches an unnanotated variant. 

719pub(crate) fn is_tag_valid_for_variant( 

720 py: pyo3::Python<'_>, 

721 tag: asn1::Tag, 

722 variant: &Variant, 

723 encoding: &Option<pyo3::Py<Encoding>>, 

724) -> bool { 

725 let ann_type = variant.ann_type.get(); 

726  

727 // There are two encodings at play here: the encoding of the CHOICE itself, 

728 // and the encoding of each of the variants. The encoding of the CHOICE will 

729 // only affect the tag if it's EXPLICIT (where it adds a wrapper). Otherwise, 

730 // we use the encoding of the variant. 

731 let encoding_to_match = match encoding { 

732 Some(e) => match e.get() { 

733 Encoding::Implicit(_) => &ann_type.annotation.get().encoding, 

734 Encoding::Explicit(_) => encoding, 

735 }, 

736 None => &ann_type.annotation.get().encoding, 

737 }; 

738  

739 is_tag_valid_for_type(py, tag, ann_type.inner.get(), encoding_to_match) 

740}

742// Given `tag` and `encoding`, returns whether that tag with that encoding 

743// matches what one would expect to see when decoding `type_` 

744pub(crate) fn is_tag_valid_for_type( 

745 py: pyo3::Python<'_>, 

746 tag: asn1::Tag, 

747 type_: &Type, 

748 encoding: &Option<pyo3::Py<Encoding>>, 

749) -> bool { 

750 match type_ { 

751 Type::Sequence(_, _) => check_tag_with_encoding(asn1::Sequence::TAG, encoding, tag), 

752 Type::Set(_, _) => check_tag_with_encoding(asn1::Set::TAG, encoding, tag), 

753 Type::SequenceOf(_) => check_tag_with_encoding(asn1::Sequence::TAG, encoding, tag), 

754 Type::SetOf(_) => check_tag_with_encoding(asn1::SetOf::<()>::TAG, encoding, tag), 

755 Type::Option(t) => is_tag_valid_for_type(py, tag, t.get().inner.get(), encoding), 

756 Type::Choice(variants) => variants.bind(py).into_iter().any(|v| { 

757 is_tag_valid_for_variant(py, tag, v.cast::<Variant>().unwrap().get(), encoding) 

758 }), 

759 Type::ValueSet(_, t, _) => is_tag_valid_for_type(py, tag, t.get().inner.get(), encoding), 

760 Type::PyBool() => check_tag_with_encoding(bool::TAG, encoding, tag), 

761 Type::PyInt() => check_tag_with_encoding(asn1::BigInt::TAG, encoding, tag), 

762 Type::PyBytes() => { 

763 check_tag_with_encoding(<&[u8] as SimpleAsn1Readable>::TAG, encoding, tag) 

764 } 

765 Type::PyStr() => check_tag_with_encoding(asn1::Utf8String::TAG, encoding, tag), 

766 Type::PrintableString() => { 

767 check_tag_with_encoding(asn1::PrintableString::TAG, encoding, tag) 

768 } 

769 Type::IA5String() => check_tag_with_encoding(asn1::IA5String::TAG, encoding, tag), 

770 Type::ObjectIdentifier() => { 

771 check_tag_with_encoding(asn1::ObjectIdentifier::TAG, encoding, tag) 

772 } 

773 Type::UtcTime() => check_tag_with_encoding(asn1::UtcTime::TAG, encoding, tag), 

774 Type::GeneralizedTime() => { 

775 check_tag_with_encoding(asn1::GeneralizedTime::TAG, encoding, tag) 

776 } 

777 Type::BitString() => check_tag_with_encoding(asn1::BitString::TAG, encoding, tag), 

778 Type::Tlv() => { 

779 match encoding { 

780 Some(e) => match e.get() { 

781 // TLVs with implicit annotations are not supported 

782 // (they are caught first at the Python level) 

783 Encoding::Implicit(_) => false, 

784 Encoding::Explicit(n) => tag == asn1::explicit_tag(*n), 

785 }, 

786 // When reading TLVs we accept any tag 

787 None => true, 

788 } 

789 } 

790 Type::Null() => check_tag_with_encoding(asn1::Null::TAG, encoding, tag), 

791 // Certificates, CSRs, CRLs, and Names are all SEQUENCEs 

792 Type::Certificate() 

793 | Type::CertificateSigningRequest() 

794 | Type::CertificateRevocationList() 

795 | Type::Name() => check_tag_with_encoding(asn1::Sequence::TAG, encoding, tag), 

796 } 

797}

822pub(crate) fn check_size_constraint( 

823 size_annotation: &Option<pyo3::Py<Size>>, 

824 data_length: impl FnOnce() -> usize, 

825 field_type: &str, 

826) -> Result<(), CryptographyError> { 

827 if let Some(size) = size_annotation { 

828 let data_length = data_length(); 

829 let min = size.get().min; 

830 let max = size.get().max.unwrap_or(usize::MAX); 

831 if !(min..=max).contains(&data_length) { 

832 return Err(CryptographyError::Py( 

833 pyo3::exceptions::PyValueError::new_err(format!( 

834 "{0} has size {1}, expected size in [{2}, {3}]", 

835 field_type, data_length, min, max 

836 )), 

837 )); 

838 } 

839 } 

840 Ok(()) 

841}
