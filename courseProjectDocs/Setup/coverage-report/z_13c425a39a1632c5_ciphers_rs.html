159 fn update<'p>( 

160 &mut self, 

161 py: pyo3::Python<'p>, 

162 data: &[u8], 

163 ) -> CryptographyResult<pyo3::Bound<'p, pyo3::types::PyBytes>> { 

164 let block_size = self.ctx.block_size(); 

165 if block_size == 1 { 

166 // Streaming modes always produce exactly data.len() bytes of 

167 // output, so encrypt directly into the result PyBytes instead 

168 // of into a zeroed scratch Vec that then gets copied. 

169 return Ok(pyo3::types::PyBytes::new_with(py, data.len(), |b| { 

170 let n = self.update_into(py, data, b)?; 

171 assert_eq!(n, data.len()); 

172 Ok(()) 

173 })?); 

174 } 

175 let mut buf = vec![0; data.len() + block_size]; 

176 let n = self.update_into(py, data, &mut buf)?; 

177 Ok(pyo3::types::PyBytes::new(py, &buf[..n])) 

178 }

198 fn update_into_inner(&mut self, data: &[u8], buf: &mut [u8]) -> CryptographyResult<usize> { 

199 let mut total_written = 0; 

200 for chunk in data.chunks(1 << 29) { 

201 // SAFETY: We ensure that outbuf is sufficiently large above. 

202 unsafe { 

203 let n = if self.is_xts { 

204 self.ctx.cipher_update_unchecked(chunk, Some(&mut buf[total_written..])).map_err(|_| { 

205 pyo3::exceptions::PyValueError::new_err( 

206 "In XTS mode you must supply at least a full block in the first update call. For AES this is 16 bytes." 

207 ) 

208 })? 

209 } else { 

210 self.ctx 

211 .cipher_update_unchecked(chunk, Some(&mut buf[total_written..]))? 

212 }; 

213 total_written += n; 

214 } 

215 } 

216  

217 Ok(total_written) 

218 }

231 pub(crate) fn finalize<'p>( 

232 &mut self, 

233 py: pyo3::Python<'p>, 

234 ) -> CryptographyResult<pyo3::Bound<'p, pyo3::types::PyBytes>> { 

235 let mut out_buf = vec![0; self.ctx.block_size()]; 

236 let n = self.ctx.cipher_final(&mut out_buf).or_else(|_| { 

237 // For authenticated modes, the only way finalize can fail is a 

238 // tag mismatch. Older OpenSSL versions (and BoringSSL, AWS-LC, 

239 // and LibreSSL) leave the error queue empty in that case, while 

240 // OpenSSL >= 4.1 pushes PROV_R_BAD_DECRYPT, so we can't key on 

241 // the error queue. 

242 if self 

243 .py_mode 

244 .bind(py) 

245 .is_instance(&types::MODE_WITH_AUTHENTICATION_TAG.get(py)?)? 

246 { 

247 return Err(CryptographyError::from(exceptions::InvalidTag::new_err(()))); 

248 } 

249 Err(CryptographyError::from( 

250 pyo3::exceptions::PyValueError::new_err( 

251 "The length of the provided data is not a multiple of the block length.", 

252 ), 

253 )) 

254 })?; 

255 Ok(pyo3::types::PyBytes::new(py, &out_buf[..n])) 

256 } 

257}

259// ChaCha20 generates 64 bytes of keystream per 32-bit block counter value. 

260const CHACHA20_BLOCK_SIZE: u64 = 64; 

261  

262// The 16-byte ChaCha20 nonce begins with a 32-bit little-endian block counter. 

263// Encrypting more than `(2**32 - counter) * 64` bytes would overflow that 

264// counter, at which point the OpenSSL implementation silently diverges from 

265// RFC 7539 (the counter carries into the rest of the nonce), so we instead 

266// refuse to encrypt past that point. 

267fn chacha20_byte_limit(nonce: &[u8]) -> u64 { 

268 let counter = u32::from_le_bytes(nonce[..4].try_into().unwrap()); 

269 ((1u64 << 32) - u64::from(counter)) * CHACHA20_BLOCK_SIZE 

270} 

271  

272fn chacha20_initial_byte_limit( 

273 py: pyo3::Python<'_>, 

274 algorithm: &pyo3::Bound<'_, pyo3::PyAny>, 

275) -> CryptographyResult<Option<u64>> { 

276 if algorithm.is_instance(&types::CHACHA20.get(py)?)? { 

277 let nonce = algorithm 

278 .getattr(pyo3::intern!(py, "nonce"))? 

279 .extract::<CffiBuf<'_>>()?; 

280 Ok(Some(chacha20_byte_limit(nonce.as_bytes()))) 

281 } else { 

282 Ok(None) 

283 } 

284}
